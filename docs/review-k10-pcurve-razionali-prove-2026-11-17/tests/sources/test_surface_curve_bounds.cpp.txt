using detail::SurfaceCurveBoundStatus;
std::vector<double> cubicKnots(const std::vector<double> &breaks) {
    std::vector<double> knots(4, breaks.front());
    for (size_t i = 1; i + 1 < breaks.size(); ++i)
        knots.insert(knots.end(), 3, breaks[i]);
    knots.insert(knots.end(), 4, breaks.back());
    return knots;
}
std::vector<double> linearControls(const std::vector<double> &breaks) {
    std::vector<double> controls{breaks.front()};
    for (size_t i = 1; i < breaks.size(); ++i)
        for (int j = 1; j <= 3; ++j)
            controls.push_back(breaks[i - 1] + (breaks[i] - breaks[i - 1]) * j / 3.0);
    return controls;
}
std::shared_ptr<BSplineSurface> tensor(const std::vector<double> &u = {0, 1},
                                     const std::vector<double> &v = {0, 1}, bool bilinear = false,
                                     double weight = 0) {
    const auto us = linearControls(u), vs = linearControls(v);
    std::vector<Vec3> poles;
    for (double x : us) for (double y : vs) poles.emplace_back(x, y, bilinear ? x * y : 0);
    return std::make_shared<BSplineSurface>(3, 3, cubicKnots(u), cubicKnots(v), int(us.size()), int(vs.size()),
        poles, weight ? std::vector<double>(poles.size(), weight) : std::vector<double>{});
}
std::shared_ptr<BSplineCurve<2>> cubicUV(std::vector<Vec2> poles = {{0, 0}, {.125, .25}, {.75, .5}, {1, 1}},
                                      Interval domain = {0, 1}, double weight = 0) {
    return std::make_shared<BSplineCurve<2>>(3,
        std::vector<double>{domain.lo, domain.lo, domain.lo, domain.lo, domain.hi, domain.hi, domain.hi, domain.hi},
        poles, weight ? std::vector<double>(poles.size(), weight) : std::vector<double>{});
}
std::shared_ptr<BSplineCurve<3>> planarEdge(const BSplineCurve<2> &uv, double weight = 0) {
    std::vector<Vec3> poles;
    for (const auto &p : uv.poles()) poles.emplace_back(p.x(), p.y(), 0);
    return std::make_shared<BSplineCurve<3>>(3, uv.knots(), poles,
        weight ? std::vector<double>(poles.size(), weight) : std::vector<double>{});
}
void certified(const CurvePtr<3> &edge, const SurfacePtr &surface, const CurvePtr<2> &uv,
               Interval range = {0, 1}, double budget = 1e-8) {
    const auto result = detail::surfaceCurveResidualBound(edge, surface, uv, range, budget);
    FK_CHECK(result.status == SurfaceCurveBoundStatus::Certified);
    FK_CHECK(std::isfinite(result.upperBound) && result.upperBound >= 0 && result.upperBound <= budget);
}
void failed(const CurvePtr<3> &edge, const SurfacePtr &surface, const CurvePtr<2> &uv,
            Interval range = {0, 1}, double budget = 1e-8) {
    FK_CHECK(detail::surfaceCurveResidualBound(edge, surface, uv, range, budget).status == SurfaceCurveBoundStatus::Failed);
}
void unsupported(const CurvePtr<3> &edge, const SurfacePtr &surface, const CurvePtr<2> &uv) {
    FK_CHECK(detail::surfaceCurveResidualBound(edge, surface, uv, {0, 1}, 1e-8).status == SurfaceCurveBoundStatus::Unsupported);
}
std::shared_ptr<BSplineCurve<3>> diagonalEdge(const std::vector<double> &breaks = {0, 1}) {
    std::vector<Vec3> poles;
    for (double t : linearControls(breaks)) poles.emplace_back(t, t, 0);
    return std::make_shared<BSplineCurve<3>>(3, cubicKnots(breaks), poles);
}
std::shared_ptr<BSplineCurve<2>> diagonalUV() {
    return std::make_shared<BSplineCurve<2>>(1, std::vector<double>{0, 0, 1, 1}, std::vector<Vec2>{{0, 0}, {1, 1}});
}
}

FK_TEST(NonIsoResidualCubicUVOnOriginalTensorPlane) {
    auto uv = cubicUV();
    certified(planarEdge(*uv), tensor(), uv);
}
FK_TEST(NonIsoResidualBilinearCarrierAndNativeLineUV) {
    auto uv = std::make_shared<Line<2>>(Vec2(0, 0), Vec2(1, 1));
    const auto d = uv->direction();
    auto edge = std::make_shared<BSplineCurve<3>>(3, cubicKnots({0, 1}),
        std::vector<Vec3>{{0, 0, 0}, {d.x()/3, d.y()/3, 0},
            {2*d.x()/3, 2*d.y()/3, d.x()*d.y()/3}, {d.x(), d.y(), d.x()*d.y()}});
    certified(edge, tensor({0, 1}, {0, 1}, true), uv);
}
FK_TEST(NonIsoResidualCommonPartitionAcrossEdgeUVAndCarrierKnots) {
    auto uv = std::make_shared<BSplineCurve<2>>(1, std::vector<double>{0, 0, .7, 1, 1},
        std::vector<Vec2>{{0, 0}, {.7, .7}, {1, 1}});
    auto edge = diagonalEdge({0, .3, 1});
    certified(edge, tensor({0, .5, 1}, {0, .5, 1}), uv, {.125, .875});
    certified(edge, tensor({0, .5, 1}, {0, .5, 1}), uv,
        {std::nextafter(.5, 0), std::nextafter(.5, 1)});
}
FK_TEST(NonIsoResidualNonUnitDomainSubrangeAndCorrelatedReversal) {
    auto uv = cubicUV({{-2, 7.5}, {-1, 8}, {2, 9}, {4, 10.5}}, {-2, 3});
    auto support = tensor({-3, 5}, {7, 11});
    certified(planarEdge(*uv), support, uv, {-1.75, 2.125});
    auto poles = uv->poles(); std::reverse(poles.begin(), poles.end());
    auto reversedUV = cubicUV(poles, {-2, 3});
    certified(planarEdge(*reversedUV), support, reversedUV, {-1.75, 2.125});
    failed(planarEdge(*uv), support, reversedUV, {-1.75, 2.125});
}
FK_TEST(NonIsoResidualConstantOriginalWeightsRemainPolynomial) {
    auto uv = cubicUV({{0, 0}, {.125, .25}, {.75, .5}, {1, 1}}, {0, 1}, 3);
    certified(planarEdge(*uv, .25), tensor({0, 1}, {0, 1}, false, 2), uv);
}
FK_TEST(NonIsoResidualHiddenEdgePeakIsNotAContinuousProof) {
    const std::vector<double> breaks{0, .201, .202, .203, 1};
    std::vector<Vec3> poles;
    for (size_t span = 0; span + 1 < breaks.size(); ++span)
        for (int j = span ? 1 : 0; j <= 3; ++j) {
            const double t = breaks[span] + (breaks[span + 1] - breaks[span]) * j / 3.0;
            const double z = span == 1 ? (j >= 2 ? 1 : 0) : span == 2 ? (j <= 1 ? 1 : 0) : 0;
            poles.emplace_back(t, t, z);
        }
    auto edge = std::make_shared<BSplineCurve<3>>(3, cubicKnots(breaks), poles);
    auto uv = diagonalUV();
    auto support = tensor();
    certified(diagonalEdge(), tensor(), uv);
    for (double t : {0., .25, .5, .75, 1.}) {
        const auto q = uv->point(t);
        FK_CHECK(norm(edge->point(t) - support->point(q.x(), q.y())) < 1e-12);
    }
    FK_CHECK(edge->point(.202).z() > .9);
    failed(edge, support, uv);
}
FK_TEST(NonIsoResidualHiddenCarrierCellPeakIsRejected) {
    const std::vector<double> breaks{0, .201, .202, .203, 1};
    const auto us = linearControls(breaks);
    std::vector<double> zs{0};
    for (size_t span = 0; span + 1 < breaks.size(); ++span)
        for (int j = 1; j <= 3; ++j)
            zs.push_back(span == 1 ? (j >= 2 ? 1 : 0) : span == 2 ? (j <= 1 ? 1 : 0) : 0);
    std::vector<Vec3> poles;
    for (size_t i = 0; i < us.size(); ++i)
        for (double v : {0., 1.0/3, 2.0/3, 1.}) poles.emplace_back(us[i], v, zs[i]);
    auto support = std::make_shared<BSplineSurface>(3, 3, cubicKnots(breaks), cubicKnots({0, 1}), int(us.size()), 4, poles);
    auto uv = diagonalUV();
    auto edge = diagonalEdge();
    certified(diagonalEdge(), tensor(), uv);
    for (double t : {0., .25, .5, .75, 1.}) {
        const auto q = uv->point(t);
        FK_CHECK(norm(edge->point(t) - support->point(q.x(), q.y())) < 1e-12);
    }
    const auto witness = uv->point(.202);
    FK_CHECK(norm(edge->point(.202) - support->point(witness.x(), witness.y())) > .9);
    failed(edge, support, uv);
}
FK_TEST(NonIsoResidualUVLeavesDomainBetweenThreeMatchingSamples) {
    auto uv = cubicUV({{0, 0}, {13.0/3, 1.0/3}, {-10.0/3, 2.0/3}, {1, 1}});
    auto support = tensor();
    auto edge = planarEdge(*uv);
    for (double t : {0., .5, 1.}) FK_CHECK(uv->point(t).x() >= -1e-14 && uv->point(t).x() <= 1 + 1e-14);
    FK_CHECK(uv->point(.25).x() > 1.3);
    failed(edge, support, uv);
}
FK_TEST(NonIsoResidualInvalidInputsFailClosed) {
    auto uv = cubicUV(); auto edge = planarEdge(*uv); auto support = tensor();
    const double nan = std::numeric_limits<double>::quiet_NaN(), inf = std::numeric_limits<double>::infinity();
    failed({}, support, uv); failed(edge, {}, uv); failed(edge, support, {});
    for (double budget : {0., -1., nan, inf}) failed(edge, support, uv, {0, 1}, budget);
    for (Interval range : {Interval{0, 0}, Interval{1, 0}, Interval{nan, 1}, Interval{0, inf}, Interval{-.01, 1}, Interval{0, 1.01}})
        failed(edge, support, uv, range);
}
FK_TEST(NonIsoResidualCurveAndCarrierGapsFailWrappersUnsupported) {
    auto uv = cubicUV(); auto edge = planarEdge(*uv); auto support = tensor();
    failed(std::make_shared<BSplineCurve<3>>(3, edge->knots(), edge->poles(), std::vector<double>{1, .25, 1, 1}), support, uv);
    failed(edge, support, std::make_shared<BSplineCurve<2>>(3, uv->knots(), uv->poles(), std::vector<double>{1, .25, 1, 1}));
    std::vector<double> weights(16, 1); weights[5] = .25;
    std::vector<Vec3> surfacePoles;
    for (int i = 0; i < 4; ++i) for (int j = 0; j < 4; ++j) surfacePoles.push_back(support->pole(i, j));
    failed(edge, std::make_shared<BSplineSurface>(3, 3, cubicKnots({0, 1}), cubicKnots({0, 1}), 4, 4,
        surfacePoles, weights), uv);
    unsupported(std::make_shared<TrimmedCurve<3>>(edge, 0, 1), support, uv);
    unsupported(edge, support, std::make_shared<TrimmedCurve<2>>(uv, 0, 1));
    certified(edge, std::make_shared<BSplineSurface>(support, Transform3()), uv, {0, 1});
    unsupported(edge, std::make_shared<Plane>(Frame3()), uv);
}
FK_TEST(NonIsoResidualCancellationPropagates) {
    auto uv = cubicUV(); auto edge = planarEdge(*uv); auto support = tensor();
    auto token = std::make_shared<std::atomic<bool>>(true); CancellationScope scope(token);
    bool cancelled = false;
    try { (void)detail::surfaceCurveResidualBound(edge, support, uv, {0, 1}, 1e-8); }
    catch (const OperationCancelled &) { cancelled = true; }
    FK_CHECK(cancelled);
}

FK_TEST(NonIsoResidualFixedInputResourceLimitFailsClosed) {
    // A valid polynomial input beyond the declared fixed 100000-control guard.
    std::vector<double> breaks;
    for (int i = 0; i <= 33334; ++i) breaks.push_back(double(i) / 33334);
    auto edge = diagonalEdge(breaks);
    FK_CHECK(edge->poleCount() > 100000);
    failed(edge, tensor(), diagonalUV());
}
