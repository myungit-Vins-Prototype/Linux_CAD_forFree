std::string number(double d) {
    std::ostringstream o;
    o << '"' << std::hexfloat << d << '"';
    return o.str();
}
template <class T> void array(std::ostream &o, const T &v) {
    o << '[';
    bool sep = false;
    for (auto x : v) {
        if (sep)
            o << ',';
        sep = true;
        o << number(x);
    }
    o << ']';
}
template <int N> void vector(std::ostream &o, const Vec<N> &v) {
    o << '[';
    for (int i = 0; i < N; ++i) {
        if (i)
            o << ',';
        o << number(v[i]);
    }
    o << ']';
}
template <int N> void curveJson(std::ostream &o, const CurvePtr<N> &c) {
    if (auto s = std::dynamic_pointer_cast<const BSplineCurve<N>>(c)) {
        o << "{\"type\":\"bspline\",\"degree\":" << s->degree()
          << ",\"knots\":";
        array(o, s->knots());
        o << ",\"poles\":[";
        for (int i = 0; i < s->poleCount(); ++i) {
            if (i)
                o << ',';
            vector(o, s->poles()[i]);
        }
        o << "],\"weights\":[";
        for (int i = 0; i < s->poleCount(); ++i) {
            if (i)
                o << ',';
            o << number(s->weight(i));
        }
        o << "]}";
    } else if (auto l = std::dynamic_pointer_cast<const Line<N>>(c)) {
        o << "{\"type\":\"line\",\"origin\":";
        vector(o, l->origin());
        o << ",\"direction\":";
        vector(o, l->direction());
        o << '}';
    } else if (auto t = std::dynamic_pointer_cast<const TrimmedCurve<N>>(c)) {
        o << "{\"type\":\"trimmed\",\"basis\":";
        curveJson(o, t->basis());
        o << '}';
    } else
        throw std::runtime_error("test export: unsupported curve wrapper");
}
void surfaceJson(std::ostream &o, const BSplineSurface &s) {
    if (s.basis())
        throw std::runtime_error(
            "test export: instance would lose placement rounding");
    o << "{\"uDegree\":" << s.uDegree() << ",\"vDegree\":" << s.vDegree()
      << ",\"uKnots\":";
    array(o, s.uKnots());
    o << ",\"vKnots\":";
    array(o, s.vKnots());
    o << ",\"uPoleCount\":" << s.uPoleCount()
      << ",\"vPoleCount\":" << s.vPoleCount() << ",\"poles\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            if (i || j)
                o << ',';
            vector(o, s.pole(i, j));
        }
    o << "],\"weights\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            if (i || j)
                o << ',';
            o << number(s.weight(i, j));
        }
    o << "]}";
}
}
namespace ForgeCad::Kernel::detail {
void traceJoin(const Body &body, FaceId face, FinId id, const SurfaceCurveBoundResult &proof, double cap, const char *phase) {
    const char *path=std::getenv("K10_JOIN_TRACE"); if (!path) return;
    const auto &fin=body.fin(id); const auto &edge=body.edge(fin.edge);
    const auto &s=dynamic_cast<const BSplineSurface &>(*body.face(face).surface);
    std::ofstream out(path,std::ios::app);
    out << "{\"phase\":\"" << phase << "\",\"face\":" << face.index << ",\"fin\":" << id.index << ",\"edge\":" << fin.edge.index
        << ",\"range\":[" << number(edge.range.lo) << ',' << number(edge.range.hi) << "],\"budget\":" << number(cap)
        << ",\"edgeTolerance\":" << number(edge.tolerance) << ",\"pcurveTolerance\":" << number(fin.pcurveTolerance)
        << ",\"status\":\"" << (proof.status==SurfaceCurveBoundStatus::Certified?"Certified":proof.status==SurfaceCurveBoundStatus::Failed?"Failed":"Unsupported")
        << "\",\"runtimeUpper\":" << number(proof.upperBound) << ",\"curve\":";
    curveJson(out,edge.curve); out << ",\"pcurve\":"; curveJson(out,fin.pcurve); out << ",\"surface\":"; surfaceJson(out,s); out << "}\n";
}
