[[noreturn]] void invalidJoin() {
    throw std::domain_error("extendSheet: contratto della giunzione non ISO modificato");
}
// Candidate construction, before the ONE continuous acceptance check.
// Source E/S, the edge parameter and the caller's fixed budget are unchanged.
// Unsupported families retain their exact original pcurve pointer.
template<int N> bool polynomialWeights(const BSplineCurve<N> &curve) {
    const double first = curve.weight(0);
    for (int i = 0; i < curve.poleCount(); ++i) {
        cancellationPoint();
        const double value = curve.weight(i);
        if (!(value > 0) || !std::isfinite(value) || value != first) return false;
    }
    return true;
}
CurvePtr<2> boundedPCurveCandidate(const Edge &edge, const SurfacePtr &surface,
                                  const CurvePtr<2> &original) {
    const auto *e = dynamic_cast<const BSplineCurve<3> *>(edge.curve.get());
    const auto *s = dynamic_cast<const BSplineSurface *>(surface.get());
    const auto *q = dynamic_cast<const BSplineCurve<2> *>(original.get());
    const auto *line = dynamic_cast<const Line<2> *>(original.get());
    if (!e || !s || (!q && !line) || e->degree() != 3
        || (q && q->degree() != 1 && q->degree() != 3)
        || (s->uDegree() != 2 && s->uDegree() != 3) || (s->vDegree() != 2 && s->vDegree() != 3)
        || s->basis() || s->isUPeriodic() || s->isVPeriodic()) return original;
    const int ne = e->poleCount(), nq = q ? q->poleCount() : 0;
    const int nu = s->uPoleCount(), nv = s->vPoleCount();
    // Bound setup BEFORE scans or copies, including a fourth carrier scalar.
    constexpr std::size_t maxScalars = 2000000;
    if (ne < 4 || ne > 100000 || (q && (nq <= q->degree() || nq > 100000))
        || nu <= s->uDegree() || nv <= s->vDegree()
        || std::size_t(nu) > maxScalars / (4 * std::size_t(nv))
        || 4 * std::size_t(nu) * std::size_t(nv) + 16 * std::size_t(ne + nq) > maxScalars
        || !edge.range.isFinite() || !(edge.range.lo < edge.range.hi)
        || !std::isfinite(edge.range.length())) return original;
    if (!polynomialWeights(*e) || (q && !polynomialWeights(*q))) return original;
    for (int i = 0; i < nu; ++i)
        for (int j = 0; j < nv; ++j) {
            cancellationPoint();
            const double weight = s->weight(i, j);
            if (!(weight > 0) || !std::isfinite(weight)) return original;
        }
    const Interval domain[] = {s->uDomain(), s->vDomain()};
    for (const auto &range : domain)
        if (!range.isFinite() || !(range.lo < range.hi)) return original;
    const auto bounded = [&](Vec2 value) {
        for (int axis = 0; axis < 2; ++axis) {
            if (!std::isfinite(value[axis])) invalidJoin();
            value[axis] = std::clamp(value[axis], domain[axis].lo, domain[axis].hi);
        }
        return value;
    };
    if (line) {
        // Always build endpoint Bernstein data: rounded endpoint containment
        // alone does not prove the exact affine expression o+d*t is inside.
        // Line would normalize the new velocity, losing the edge parameter.
        const auto first = bounded(line->origin() + line->direction() * edge.range.lo);
        const auto last = bounded(line->origin() + line->direction() * edge.range.hi);
        return std::make_shared<BSplineCurve<2>>(1,
            std::vector<double>{edge.range.lo, edge.range.lo, edge.range.hi, edge.range.hi},
            std::vector<Vec2>{first, last});
    }
    bool changed = false;
    for (const auto &pole : q->poles()) {
        cancellationPoint();
        const auto candidate = bounded(pole);
        changed = changed || candidate[0] != pole[0] || candidate[1] != pole[1];
    }
    if (!changed) return original;
    std::vector<Vec2> poles;
    poles.reserve(std::size_t(nq));
    for (const auto &pole : q->poles()) {
        cancellationPoint();
        poles.push_back(bounded(pole));
    }
    return std::make_shared<BSplineCurve<2>>(q->degree(), q->knots(), std::move(poles), q->weights());
}
void requireResult(const SurfaceCurveBoundResult &result, double budget) {
    if (result.status != SurfaceCurveBoundStatus::Certified
        || !std::isfinite(result.upperBound) || !(result.upperBound >= 0.0)
        || result.upperBound > budget)
        throw std::domain_error("extendSheet: giunzione non ISO fuori dal budget continuo");
}
}

std::vector<EmittedSheetJoin> certifySheetJoins(Body &body,
    const std::vector<SheetJoinPlan> &plans, const std::vector<EdgeId> &modelEdges) {
    const auto faces = body.faces();
    std::vector<EmittedSheetJoin> result;
    std::set<std::pair<int, int>> incidences;
    for (const auto &plan : plans) {
        cancellationPoint();
        if (plan.face < 0 || std::size_t(plan.face) >= faces.size()
            || plan.edge < 0 || std::size_t(plan.edge) >= modelEdges.size()
            || !std::isfinite(plan.budget) || !(plan.budget > 0.0)) invalidJoin();
        const FaceId face = faces[std::size_t(plan.face)];
        const EdgeId edgeId = modelEdges[std::size_t(plan.edge)];
        if (!body.contains(edgeId) || !incidences.emplace(face.index, edgeId.index).second) invalidJoin();
        const Edge &edge = body.edge(edgeId);
        if (edge.tolerance != plan.budget || edge.curve != plan.curve
            || edge.range.lo != plan.range.lo || edge.range.hi != plan.range.hi) invalidJoin();
        int uses = 0;
        for (LoopId loop : body.face(face).loops)
            for (FinId finId : body.loopFins(loop)) {
                Fin &fin = body.fin(finId);
                if (fin.edge != edgeId) continue;
                ++uses;
                if (!edge.curve || !body.face(face).surface || !fin.pcurve) invalidJoin();
                const auto candidate = boundedPCurveCandidate(edge, body.face(face).surface, fin.pcurve);
                const auto proof = surfaceCurveResidualBound(edge.curve, body.face(face).surface,
                    candidate, edge.range, plan.budget);
                if (proof.status == SurfaceCurveBoundStatus::Unsupported) continue;
                requireResult(proof, plan.budget);
                fin.pcurve = candidate;
                fin.pcurveTolerance = plan.budget;
                traceJoin(body, face, finId, proof, plan.budget, "post-build");
                result.push_back({face, edgeId, edge.curve, fin.pcurve, edge.range, plan.budget, fin.sense});
            }
        if (uses != 1) invalidJoin();
    }
    return result;
}

void requireUnifiedSheetJoins(const Body &body, const BodyHistory &history,
    const std::vector<EmittedSheetJoin> &plans) {
    std::set<int> incidences;
    for (const auto &plan : plans) {
        cancellationPoint();
        const auto faces = history.descendants(EntityKind::Face, 0, plan.face.index);
        const auto edges = history.descendants(EntityKind::Edge, 0, plan.edge.index);
        int uses = 0;
        for (int faceIndex : faces) {
            const FaceId face(faceIndex);
            if (!body.contains(face)) invalidJoin();
            for (LoopId loop : body.face(face).loops)
                for (FinId finId : body.loopFins(loop)) {
                    const Fin &fin = body.fin(finId);
                    if (std::find(edges.begin(), edges.end(), fin.edge.index) == edges.end()) continue;
                    const Edge &edge = body.edge(fin.edge);
                    if (!incidences.insert(finId.index).second || edge.curve != plan.curve
                        || fin.pcurve != plan.pcurve || fin.sense != plan.sense
                        || edge.range.lo != plan.range.lo || edge.range.hi != plan.range.hi
                        || edge.tolerance != plan.budget || fin.pcurveTolerance != plan.budget) invalidJoin();
                    const auto proof = surfaceCurveResidualBound(edge.curve, body.face(face).surface,
                        fin.pcurve, edge.range, plan.budget);
                    requireResult(proof, plan.budget);
                    traceJoin(body, face, finId, proof, plan.budget, "post-unify");
                    ++uses;
                }
        }
        if (uses != 1) invalidJoin();
    }
}
