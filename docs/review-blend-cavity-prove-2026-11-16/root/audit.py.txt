def copy_folder(label,folder):
    for p in folder.rglob('*'):
        if not p.is_file() or '__pycache__' in p.parts or p.suffix in ['.o','.a','.body','.prt','.pyc']:continue
        with p.open('rb') as stream:
            if stream.read(4)==b'\x7fELF':continue
        q=P/label/p.relative_to(folder)
        if q.suffix in ['.cpp','.h','.py','.sh'] or q.name=='CMakeLists.txt':q=q.with_name(q.name+'.txt')
        q.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(p,q)
    # Adapt embedded manifests when proof sources gain the archival .txt suffix.
    for manifest in folder.rglob('SHA256SUMS'):
        target=P/label/manifest.relative_to(folder)
        lines=[]
        for line in manifest.read_text().splitlines():
            h,f=line.split('  ',1);assert sha(manifest.parent/f)==h,(label,f)
            archived=Path(f)
            if archived.suffix in ['.cpp','.h','.py','.sh'] or archived.name=='CMakeLists.txt':archived=archived.with_name(archived.name+'.txt')
            assert sha(target.parent/archived)==h,(label,archived)
            lines.append(h+'  '+str(archived)+'\n')
        if ''.join(lines)!=manifest.read_text():
            target.with_name('SHA256SUMS.original.txt').write_bytes(manifest.read_bytes())
            target.write_text(''.join(lines))
