#!/usr/bin/env python3
"""Independent exact whole-carrier separation, starting from frozen IEEE B-spline inputs.
No production projector result is trusted as an error bound. The exported projection
supplies only an arbitrary rational test direction for a convex-hull lower bound.
"""
from fractions import Fraction as F
from pathlib import Path
from math import isqrt
import json, hashlib, sys, time
P=Path(__file__).parent
source=P/'failure-data.json'
t0=time.monotonic()
j=json.loads(source.read_text())
def exact(s): return F.from_float(float.fromhex(s))
def blend(a,b,t): return [(1-t)*x+t*y for x,y in zip(a,b)]
def evaluate(c,t):
    d=c['degree'];U=list(map(exact,c['knots']));W=list(map(exact,c['weights'])) or [F(1)]*len(c['poles'])
    Q=[[exact(x)*w for x in p]+[w] for p,w in zip(c['poles'],W)]
    assert len(U)==len(Q)+d+1 and min(W)>0
    k=next(k for k in range(d,len(Q)) if U[k]<=t<U[k+1])
    a=Q[k-d:k+1]
    for r in range(1,d+1):
        for m in range(d,r-1,-1):
            i=k-d+m; alpha=(t-U[i])/(U[i+d-r+1]-U[i]);a[m]=blend(a[m-1],a[m],alpha)
    assert a[d][3]>0
    return [x/a[d][3] for x in a[d][:3]]
point=evaluate(j['curve'],exact(j['parameter']))
normal=[x-exact(y) for x,y in zip(point,j['projectedPoint'])]
norm2=sum(x*x for x in normal);assert norm2>0
budget=exact(j['tolerance']);threshold=10*budget;threshold2=threshold*threshold
s=j['surface'];pu=s['uDegree'];pv=s['vDegree'];U=list(map(exact,s['uKnots']));V=list(map(exact,s['vKnots']))
nu=len(s['poles']);nv=len(s['poles'][0]);assert len(U)==nu+pu+1 and len(V)==nv+pv+1
assert U[:pu+1]==[U[pu]]*(pu+1) and U[nu:]==[U[nu]]*(pu+1)
assert V[:pv+1]==[V[pv]]*(pv+1) and V[nv:]==[V[nv]]*(pv+1)
# Each v control is a vector of all homogeneous u controls. Tensor insertion is exact.
controls=[]
for v in range(nv):
    row=[]
    for u in range(nu):
        w=exact(s['weights'][u][v]);assert w>0
        row.extend([exact(x)*w for x in s['poles'][u][v]]+[w])
    controls.append(row)
insertions=0
for knot in sorted(set(V[pv+1:len(controls)])):
    while V.count(knot)<pv:
        n=len(controls)-1; mult=V.count(knot)
        k=next(i for i in range(pv,n+1) if V[i]<=knot<V[i+1])
        out=[None]*(n+2)
        for i in range(k-pv+1):out[i]=controls[i]
        for i in range(k-mult+1,n+2):out[i]=controls[i-1]
        for i in range(k-pv+1,k-mult+1):out[i]=blend(controls[i-1],controls[i],(knot-V[i])/(V[i+pv]-V[i]))
        assert all(x is not None for x in out)
        controls=out;V=V[:k+1]+[knot]+V[k+1:];insertions+=1
assert all(U.count(x)>=pu for x in set(U[pu+1:nu]))
us=[k for k in range(pu,nu) if U[k]<U[k+1]]
vs=[k for k in range(pv,len(controls)) if V[k]<V[k+1]]
patches=[]
for ku in us:
    for kv in vs:
        patch=[[controls[v][4*u:4*u+4] for v in range(kv-pv,kv+1)] for u in range(ku-pu,ku+1)]
        patches.append((patch,0,0))
print(json.dumps({'stage':'exact_extraction','patches':len(patches),'insertions':insertions,'seconds':time.monotonic()-t0}),flush=True)
def lower2(patch):
    poles=[[h[a]/h[3] for a in range(3)] for row in patch for h in row]
    assert all(h[3]>0 for row in patch for h in row)
    low=[];high=[]
    for a in range(3):low.append(min(q[a] for q in poles));high.append(max(q[a] for q in poles))
    box=sum((low[a]-point[a] if point[a]<low[a] else point[a]-high[a] if point[a]>high[a] else F(0))**2 for a in range(3))
    support=min(sum(normal[a]*(point[a]-q[a]) for a in range(3)) for q in poles)
    support=support*support/norm2 if support>0 else F(0)
    return max(box,support)
def split_curve(c):
    left=[c[0]];right=[c[-1]]
    while len(c)>1:
        c=[blend(a,b,F(1,2)) for a,b in zip(c,c[1:])];left.append(c[0]);right.append(c[-1])
    return left,right[::-1]
def subdivide(patch,axis):
    if axis==1:
        halves=[split_curve(row) for row in patch]
        return [[h[0] for h in halves],[h[1] for h in halves]]
    trans=list(map(list,zip(*patch)));halves=[split_curve(row) for row in trans]
    return [list(map(list,zip(*(h[0] for h in halves)))),list(map(list,zip(*(h[1] for h in halves))))]
def spread(patch,axis):
    rows=patch if axis==1 else list(zip(*patch))
    return max(sum((float(r[-1][a]/r[-1][3])-float(r[0][a]/r[0][3]))**2 for a in range(3)) for r in rows)
# Sensitivity: a genuine carrier corner must block whole-carrier separation.
original_point=point
first_h=patches[0][0][0][0]
point=[x/first_h[3] for x in first_h[:3]]
assert point==list(map(exact,s['poles'][0][0]))
assert lower2(patches[0][0])==0
point=original_point
initial=len(patches);visited=leaves=0;max_depth=0;minimum=None
while patches:
    patch,du,dv=patches.pop();visited+=1
    lb=lower2(patch)
    if lb>threshold2:
        leaves+=1;minimum=lb if minimum is None else min(minimum,lb);max_depth=max(max_depth,du+dv);continue
    if visited>100000 or du+dv>=60:raise RuntimeError(f'unresolved work limit {visited} {du} {dv}')
    # Float is only a subdivision-order heuristic, never an acceptance test.
    axis=0 if spread(patch,0)>spread(patch,1) else 1
    if du>=40:axis=1
    if dv>=40:axis=0
    for half in subdivide(patch,axis):patches.append((half,du+(axis==0),dv+(axis==1)))
    if visited%1000==0:print(json.dumps({'stage':'subdivision','visited':visited,'pending':len(patches)}),flush=True)
assert minimum>threshold2 and leaves>0
# Downward-rounded exact square-root bound, then decimal shown only as illustration.
bits=128;k=isqrt((minimum.numerator<<(2*bits))//minimum.denominator);lo=F(k,1<<bits);assert lo*lo<=minimum and lo>threshold
out={'pass':True,'scope':'all closed parameter cells of frozen original carrier, exact rational arithmetic; single bad curve point only',
 'input_sha256':hashlib.sha256(source.read_bytes()).hexdigest(),'curve_point':[str(x) for x in point],
 'edge_budget_exact':str(budget),'threshold_exact':str(threshold),'global_lower_bound_exact':str(lo),
 'global_lower_bound_display_mm':float(lo),'threshold_display_mm':float(threshold),
 'on_carrier_corner_sensitivity':True,'clamped_endpoint_checks':True,
 'initial_patches':initial,'visited':visited,'accepted_leaves':leaves,'max_depth':max_depth,
 'v_knot_insertions':insertions,'seconds':time.monotonic()-t0,
 'limitations':'Frozen intermediate serialized B-rep coefficients, not a certificate for every point of either old or corrected curve. Test direction is arbitrary; no projector optimality assumed.'}
(P/'fraction-separation-result.json').write_text(json.dumps(out,indent=2)+'\n');print(json.dumps(out),flush=True)
