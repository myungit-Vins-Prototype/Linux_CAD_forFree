from pathlib import Path
import os,subprocess,json,time
p=Path('/tmp/blend-cavity-review/tests');env=os.environ.copy();env['ASAN_OPTIONS']='detect_leaks=1:halt_on_error=1';env['UBSAN_OPTIONS']='halt_on_error=1:print_stacktrace=1';env.pop('LSAN_OPTIONS',None)
meta={'scope':'ASan/UBSan/LSan targeted four compiled source TUs; remaining kernel Release; no whole-kernel memory certificate','ASAN_OPTIONS':env['ASAN_OPTIONS'],'UBSAN_OPTIONS':env['UBSAN_OPTIONS'],'LSAN_OPTIONS':None,'runs':{}}
for name in ['probe','regressions']:
 start=time.monotonic()
 with (p/('sanitizer-'+name+'-final.log')).open('w') as f:result=subprocess.run([str(p/'sanitizer'/name)],env=env,stdout=f,stderr=subprocess.STDOUT)
 meta['runs'][name]={'exit':result.returncode,'seconds':time.monotonic()-start}
(p/'sanitizer-run-final.json').write_text(json.dumps(meta,indent=2)+'\n')
print(json.dumps(meta))
