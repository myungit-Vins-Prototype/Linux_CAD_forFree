import json,hashlib

plan={'files':items,'bytes':sum(x['bytes'] for x in items),'scope':'only own completed /tmp/k10-quadrature-review/root staging; production build, sources, logs and CAD preserved'};p=T/'cleanup.json';assert not p.exists();p.write_text(json.dumps(plan,indent=2)+'\n')
for x in items:
 p=T/x['path'];assert hashlib.sha256(p.read_bytes()).hexdigest()==x['sha256'];p.unlink()
