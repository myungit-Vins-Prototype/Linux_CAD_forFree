from pathlib import Path
import json,hashlib,subprocess,time,os,sys
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/k10-quadrature-review/root');K=R/'kernel';lib=R/'forgecad-release/kernel/libforgekernel.a';sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def run(name,args,allow=(0,),timeout=600,env=None):
 start=time.monotonic()
 with (T/(name+'.log')).open('w') as out:code=subprocess.run(['timeout','--kill-after=2',str(timeout),'/usr/bin/time','-v','-o',str(T/(name+'-time.txt'))]+list(map(str,args)),cwd=R,stdout=out,stderr=subprocess.STDOUT,env=env).returncode
 j={'command':list(map(str,args)),'exit':code,'seconds':time.monotonic()-start,'watchdog_seconds':timeout};(T/(name+'-result.json')).write_text(json.dumps(j,indent=2)+'\n');print(name,code,round(j['seconds'],3),flush=True)
 if code not in allow:print((T/(name+'.log')).read_text()[-5000:]);raise SystemExit(1)
 return j
flags=['/usr/bin/c++','-std=c++17','-O2','-g','-fno-fast-math','-ffp-contract=off','-Wall','-Wextra','-Wpedantic','-I'+str(K),'-I'+str(K/'tests')]

C=Path('/tmp/k10-quadrature-review/core/v2/kernel');N=Path('/tmp/k10-quadrature-review/tests-v2/test_quadrature_validation.cpp');cf=flags[:1]+['-I'+str(C)]+flags[1:]

run('compile-candidate-mass',cf+['-c',C/'fk_mass.cpp','-o',T/'mass.o'])
for label,fl,extra in [('baseline',flags,[]),('candidate',cf,[T/'candidate.o',T/'mass.o'])]:
 run('link-'+label+'-mass-golden',fl+[T/'ordinary-mass.cpp']+extra+[lib,'-pthread','-o',T/(label+'-mass-golden')])
 run(label+'-mass-golden',[T/(label+'-mass-golden')],timeout=30)
assert (T/'baseline-mass-golden.log').read_bytes()==(T/'candidate-mass-golden.log').read_bytes()
run('link-candidate-length-golden',cf+[T/'ordinary-length.cpp',T/'candidate.o',lib,'-pthread','-o',T/'candidate-length-golden'])
run('candidate-length-golden',[T/'candidate-length-golden'],timeout=10)
assert (T/'baseline-length-golden.log').read_bytes()==(T/'candidate-length-golden.log').read_bytes()
