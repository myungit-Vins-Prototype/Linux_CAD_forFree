#include "cad_drawing_curve_recognition.h"
#include <atomic>
#include <thread>
#include <iostream>
using namespace ForgeCad;
using namespace ForgeCad::Kernel;
int main(){
 auto base=conicArcToBSpline<3>(Vec3(0,0,0),Vec3(10,0,0),Vec3(0,10,0),0,kTwoPi);
 for(int i=1;i<500;i++){double t=kTwoPi*i/500; if(i%125)base=base.insertKnot(t);}
 for(int round=0;round<200;round++){
  const auto curve=std::make_shared<const BSplineCurve<3>>(base);std::atomic<int> ready{0};std::atomic<bool> go{false};std::vector<std::thread> jobs;
  for(int i=0;i<8;i++)jobs.emplace_back([&]{ready++;while(!go.load()){};auto circle=drawingRecognizeCircle(*curve,1e-6);if(!circle)std::abort();});
  while(ready!=8){};go=true;for(auto &job:jobs)job.join();
  if(round%10==0)std::cerr<<round<<"\n";
 }
}
