#include "fk_offset.h"
#include "fk_cancel.h"
#include "fk_surface.h"
#include <cmath>
#include <iomanip>
#include <iostream>
using namespace ForgeCad::Kernel;
int main(){
 std::cout<<std::setprecision(16);
 struct Case{const char *name; double R,r,d,lo,hi;};
 for(const Case c:{Case{"ring_valid",5,1,3,0,kTwoPi},Case{"ring_horn_full",5,1,4,0,kTwoPi},Case{"ring_spindle_full",5,1,5,0,kTwoPi},Case{"ring_outer_trim",5,1,5,-.4,.4},Case{"ring_outer_shifted",5,1,5,2*kTwoPi-.4,2*kTwoPi+.4},Case{"ring_inner_shifted",5,1,5,3*kPi-.2,3*kPi+.2},Case{"spindle_outer",5,6,.1,-.3,.3},Case{"spindle_inner",5,6,.1,kPi-.15,kPi+.15},Case{"horn_outer",5,5,.1,-.3,.3},Case{"horn_full",5,5,-.1,0,kTwoPi}}){
  const ToroidalSurface source(Frame3(),c.R,c.r);
  try{
   const auto out=offsetSurface(source,c.d,{.1,.8},{c.lo,c.hi},1e-7);
   double maximum=0,minimumDot=1;
   for(int i=0;i<=100;++i){const double v=c.lo+(c.hi-c.lo)*i/100.;try{maximum=std::max(maximum,distance(out->point(.4,v),source.point(.4,v)+c.d*source.normal(.4,v)));minimumDot=std::min(minimumDot,dot(out->normal(.4,v),source.normal(.4,v)));}catch(const std::domain_error&){minimumDot=-2;}}
   std::cout<<c.name<<" accepted=1 error="<<maximum<<" min_normal_dot="<<minimumDot<<"\n";
  }catch(const std::exception&e){std::cout<<c.name<<" accepted=0 reason="<<e.what()<<"\n";}
 }
 for(bool initially:{false,true}){
  auto token=std::make_shared<std::atomic<bool>>(initially);CancellationScope scope(token);int calls=0;double deviation=-123;
  try{auto result=fitCurve([&](double t){++calls;if(t==.75)token->store(true);return Vec3(t,0,0);},{0,1},{},1e-7,&deviation);std::cout<<"cancel initially="<<initially<<" returned=1 calls="<<calls<<" deviation="<<deviation<<"\n";}
  catch(const OperationCancelled&){std::cout<<"cancel initially="<<initially<<" cancelled=1 calls="<<calls<<" deviation="<<deviation<<"\n";}
 }
 {
  const auto f=[](double t){double q=1;for(double r:{0.,.25,.5,.75,1.})q*=t-r;return Vec3(t,0,1e4*q*q*q);};
  double measured=-1;const auto result=fitCurve(f,{0,1},{},1e-7,&measured);double actual=0;
  for(int i=0;i<=10000;++i){const double t=i/10000.;actual=std::max(actual,distance(result->point(t),f(t)));}
  std::cout<<"fit_hidden_polynomial measured="<<measured<<" actual="<<actual<<" poles="<<result->poleCount()<<"\n";
 }
 {
  const std::vector<double> z={0,0,0,0.0023926594471666389,-0.03158506535957388,0.22274247758415189,-1.0588644597919092,3.7068614333740251,-10.150702528319336,22.626599248232065,-42.105538437516088,66.431608164324672,-89.673893632054543,104.07526656293439,-104.07526656293439,89.673893632054543,-66.431608164324672,42.105538437516088,-22.626599248232065,10.150702528319336,-3.7068614333740251,1.0588644597919092,-0.22274247758415189,0.03158506535957388,-0.0023926594471666389,0,0,0};std::vector<Vec3> controls;for(int i=0;i<=27;++i)controls.emplace_back(i/27.,0,z[i]);
  std::vector<double> knots(28,0);knots.insert(knots.end(),28,1.);
  auto base=std::make_shared<BSplineCurve<3>>(27,knots,controls);ExtrusionSurface source(base,Vec3(0,1,0));
  double pe=0,de=0,dde=0;
  for(int i=0;i<=10000;++i){const long double t=i/10000.L;long double p=1,q=0,r=0;
   for(long double root:{0.L,1.L,.25L,.5L,.75L,.05L,.275L,.725L,.95L})for(int j=0;j<3;++j){r=r*(t-root)+2*q;q=q*(t-root)+p;p=p*(t-root);}
   Vec3 jet[3];base->evaluate(double(t),2,jet);pe=std::max(pe,distance(jet[0],Vec3(double(t),0,double(1e10L*p))));de=std::max(de,distance(jet[1],Vec3(1,0,double(1e10L*q))));dde=std::max(dde,distance(jet[2],Vec3(0,0,double(1e10L*r))));}
  std::cout<<"source_evaluation max_position_error="<<pe<<" max_first_derivative_error="<<de<<" max_second_derivative_error="<<dde<<"\n";
  try{const auto result=offsetSurface(source,.001,{0,1},{0,1},1e-7);double actual=0,originalPeak=0,minRadius=1e100;
   for(int i=0;i<=10000;++i){double t=i/10000.;actual=std::max(actual,distance(result->point(t,.5),source.point(t,.5)+.001*source.normal(t,.5)));Vec3 jet[3];base->evaluate(t,2,jet);originalPeak=std::max(originalPeak,std::abs(jet[0].z()));const double speed=norm(jet[1]);const double crossnorm=norm(cross(jet[1],jet[2]));if(crossnorm>0)minRadius=std::min(minRadius,speed*speed*speed/crossnorm);}
   std::cout<<"real_spline_degree27 accepted=1 actual_error="<<actual<<" original_peak="<<originalPeak<<" minimum_sampled_curvature_radius="<<minRadius<<"\n";
  }catch(const std::exception&e){std::cout<<"real_spline_degree27 accepted=0 reason="<<e.what()<<"\n";}
 }

}
