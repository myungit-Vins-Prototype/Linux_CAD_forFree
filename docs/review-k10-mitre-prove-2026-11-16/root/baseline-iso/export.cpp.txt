#include "json_helpers.inc"
#include <iostream>
std::shared_ptr<BSplineSurface> hiddenIso(bool bump) {
    const double a = 1.0 / 512.0, b = 3.0 / 512.0;
    const std::vector<double> y{0, a / 2, a, (a + b) / 2, b, (b + 1) / 2, 1};
    std::vector<Vec3> poles;
    for (int i = 0; i < 2; ++i)
        for (std::size_t j = 0; j < y.size(); ++j)
            poles.push_back({double(i), y[j], bump && j == 3 ? 1.0 : 0.0});
    return std::make_shared<BSplineSurface>(1, 2, std::vector<double>{0,0,1,1},
        std::vector<double>{0,0,0,a,a,b,b,1,1,1}, 2, int(y.size()), poles);
}

int main() {
    const auto s = hiddenIso(false), t = hiddenIso(true);
    const double end[4]{0, 1, 0, 1};
    const auto cut = detail::traceBlendMitre(*s, *t, 0, 0, end, 1, 1e-9);
    const CurvePtr<2> uv = std::make_shared<BSplineCurve<2>>(1, std::vector<double>{0,0,1,1}, std::vector<Vec2>{{0,0},{0,1}});
    std::ofstream out("/tmp/mitre-review/root/baseline-iso/coefficients.json"); out << std::setprecision(17);
    out << "{\"scope\":\"baseline internal ISO cut versus original affine maps, NOT Body fins\",\"surface_s\":";
    surfaceJson(out, *s); out << ",\"surface_t\":"; surfaceJson(out, *t);
    out << ",\"curve\":"; curveJson(out, cut.curve);
    out << ",\"pcurve_s\":"; curveJson(out, uv); out << ",\"pcurve_t\":"; curveJson(out, uv);
    out << ",\"range\":[" << cut.range.lo << ',' << cut.range.hi << "],\"reported_gap\":" << cut.gap << ",\"budget\":" << linearResolution() << "}\n";
    std::cout << "baseline ISO returned gap=" << std::setprecision(17) << cut.gap << std::endl;
}
