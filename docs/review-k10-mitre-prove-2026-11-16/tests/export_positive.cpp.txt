#include "fk_blend_mitre.h"
#include "fk_bspline.h"
#include "fk_precision.h"
#include <fstream>
#include <iomanip>
#include <iostream>
using namespace ForgeCad::Kernel;
void surfaceJson(std::ostream &out, const BSplineSurface &s) {
    out << "{\"u_degree\":" << s.uDegree() << ",\"v_degree\":" << s.vDegree()
        << ",\"u_count\":" << s.uPoleCount() << ",\"v_count\":" << s.vPoleCount() << ",\"u_knots\":[";
    for (std::size_t i = 0; i < s.uKnots().size(); ++i) out << (i ? "," : "") << s.uKnots()[i];
    out << "],\"v_knots\":[";
    for (std::size_t i = 0; i < s.vKnots().size(); ++i) out << (i ? "," : "") << s.vKnots()[i];
    out << "],\"poles\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            const Vec3 p = s.pole(i, j);
            out << (i || j ? "," : "") << '[' << p.x() << ',' << p.y() << ',' << p.z() << ']';
        }
    out << "],\"weights\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j)
            out << (i || j ? "," : "") << s.weight(i, j);
    out << "]}";
}
template<int N> void curveJson(std::ostream &out, const CurvePtr<N> &ptr) {
    const auto &c = dynamic_cast<const BSplineCurve<N> &>(*ptr);
    out << "{\"degree\":" << c.degree() << ",\"knots\":[";
    for (std::size_t i = 0; i < c.knots().size(); ++i) out << (i ? "," : "") << c.knots()[i];
    out << "],\"poles\":[";
    for (std::size_t i = 0; i < c.poles().size(); ++i) {
        out << (i ? "," : "") << '[';
        for (int k = 0; k < N; ++k) out << (k ? "," : "") << c.poles()[i][k];
        out << ']';
    }
    out << "],\"weights\":[";
    for (std::size_t i = 0; i < c.weights().size(); ++i) out << (i ? "," : "") << c.weights()[i];
    out << "]}";
}
int main() {
    std::ofstream out("/tmp/mitre-review/tests/candidate-positive-coefficients.json");
    out << std::setprecision(17) << "[";
    for (int variant = 0; variant < 5; ++variant) {
        const double scale = variant == 3 ? 2.5 : 1.;
        const Transform3 tr = variant >= 2 && variant <= 3
            ? Transform3::translation({7, -4, 2}) * Transform3::rotation({}, Vec3(1, 2, 3), .42) * Transform3::scaling({}, scale)
            : Transform3::scaling({}, scale);
        std::vector<Vec3> poles, other;
        std::vector<double> weights;
        for (int i = 0; i < 2; ++i)
            for (int j = 0; j < 3; ++j) {
                poles.push_back(tr.applyToPoint({double(i), j / 2., 0}));
                other.push_back(tr.applyToPoint({double(i), j / 2., variant == 4 ? .5 * linearResolution() : 0}));
                weights.push_back(variant >= 1 && variant <= 3 && j == 1 ? .7 : 1.);
            }
        const auto s = std::make_shared<BSplineSurface>(1, 2, std::vector<double>{0,0,1,1}, std::vector<double>{0,0,0,1,1,1}, 2, 3, poles, weights);
        const auto t = std::make_shared<BSplineSurface>(1, 2, std::vector<double>{0,0,1,1}, std::vector<double>{0,0,0,1,1,1}, 2, 3, other, weights);
        const double endpoint[4]{.3, 1, .3, 1};
        const auto cut = detail::traceBlendMitre(*s, *t, .3, .3, endpoint, scale, 1e-9);
        out << (variant ? "," : "") << "{\"variant\":" << variant << ",\"surface_s\":";
        surfaceJson(out, *s); out << ",\"surface_t\":"; surfaceJson(out, *t);
        out << ",\"curve\":"; curveJson(out, cut.curve);
        out << ",\"pcurve_s\":"; curveJson(out, cut.pcurveS);
        out << ",\"pcurve_t\":"; curveJson(out, cut.pcurveT);
        out << ",\"range\":[" << cut.range.lo << ',' << cut.range.hi << "],\"reported_gap\":" << cut.gap
            << ",\"budget\":" << cut.certificateBudget << "}";
        std::cout << "variant=" << variant << " budget=" << std::setprecision(17) << cut.certificateBudget << " gap=" << cut.gap << '\n';
    }
    out << "]\n";
}
