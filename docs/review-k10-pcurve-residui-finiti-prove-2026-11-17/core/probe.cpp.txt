#include "fk_pcurve.h"
#include "fk_surface.h"
#include <iostream>
#include <limits>
using namespace ForgeCad::Kernel;
int main(){
 const Plane s{Frame3()}; const auto c=std::make_shared<Line<3>>(Vec3(0,0,1),Vec3(1,0,0));
 const Line<2> q(Vec2(),Vec2(1,0));
 for(int samples:{-1,0,1,16})std::cout<<"samples="<<samples<<" deviation="<<pcurveDeviation(s,*c,q,{0,1},samples)<<'\n';
 const double big=std::numeric_limits<double>::max();
 const auto large=std::make_shared<Circle<3>>(Vec3(big,0,0),Vec3(1,0,0),Vec3(0,1,0),.5*big);
 const Circle<2> uv(Vec2(big,0),Vec2(1,0),Vec2(0,1),.5*big);
 std::cout<<"nonfinitepoint="<<!isFinite(large->point(0))<<" deviation="<<pcurveDeviation(s,*large,uv,{0,.1})<<" exact="<<bool(exactPCurve(s,large,{0,.1},1e-8))<<'\n';
}
