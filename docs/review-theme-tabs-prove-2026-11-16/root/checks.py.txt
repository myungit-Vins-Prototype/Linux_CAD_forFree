from pathlib import Path
import subprocess,json,time,re,sys,os
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/theme-tabs-review/root')
def run(name,cmd,timeout=None,env=None):
    start=time.monotonic()
    with (T/(name+'.log')).open('w') as log:
        try:code=subprocess.run(cmd,cwd=R,stdout=log,stderr=subprocess.STDOUT,timeout=timeout,env=env).returncode
        except subprocess.TimeoutExpired:code=124
    result=dict(exit=code,seconds=time.monotonic()-start,command=cmd)
    (T/(name+'-result.json')).write_text(json.dumps(result,indent=2)+'\n');print(name,result,flush=True)
    if code:
        print((T/(name+'.log')).read_text()[-5000:],flush=True);raise SystemExit(code)
    return result
if __name__=='__main__':
    if sys.argv[1]=='build':run('build',['cmake','--build','forgecad-release','--parallel','2'])
    elif sys.argv[1]=='tests':
        # Direct callers, topological contracts, cancellation/preview and the
        # exact CTest selection from the preceding phase that exposed this bug.
        wanted=['forgecad_document_theme_tabs_tests','forgecad_document_theme_only2d_tests','forgecad_document_tabs_tests','forgecad_application_theme_tests','forgecad_function_panels_tests','forgecad_gui_3d_review_tests','forgecad_storyboard_review_tests','forgecad_storyboard_panel_selection_tests','forgecad_fillet_selection_review_tests','forgecad_chamfer_preview_state_tests','forgecad_preview_selection_state_tests','forgecad_history_suppression_tests','forgecad_drawing_ui_controls_tests','forgecad_drawing_page_tests','forgecad_drawing_multisheet_tests','forgecad_drawing_revisions_tests','forgecad_drawing_restore_tests','forgecad_drawing_release_tests','forgecad_ellipse_extremes_tests','forgecad_blend_history_tests','forgekernel_blend_cavity_regression_tests']
        actual={t['name'] for t in json.loads(subprocess.check_output(['ctest','--test-dir','forgecad-release','--show-only=json-v1'],cwd=R))['tests']}
        assert set(wanted)<=actual,set(wanted)-actual
        result=run('ctest',['ctest','--test-dir','forgecad-release','--output-on-failure','--parallel','1','-V','-R','^('+'|'.join(map(re.escape,wanted))+')$'])
        result.update(tests=wanted,actual_passed=len(wanted));(T/'ctest-result.json').write_text(json.dumps(result,indent=2)+'\n')
    elif sys.argv[1]=='kernel':
        run('blend',['forgecad-release/kernel/forgekernel_tests','Blend'],timeout=240)
        run('helix',['forgecad-release/kernel/forgekernel_tests','Helix'],timeout=240)
        run('fill',['forgecad-release/kernel/forgekernel_tests','Fill'],timeout=240)
