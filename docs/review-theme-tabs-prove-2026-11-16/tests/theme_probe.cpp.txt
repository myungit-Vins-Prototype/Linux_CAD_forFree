static int checks=0,failures=0;
static QString output;
static void check(bool value,const QString &name){++checks;if(!value){++failures;std::cout<<"FAIL "<<name.toStdString()<<std::endl;}}
static void ok(const QString &s){if(!s.isEmpty())throw std::runtime_error(s.toStdString());}
static void settle(){for(int i=0;i<8;++i)QApplication::processEvents(QEventLoop::AllEvents,20);}
static QColor dominant(QWidget *w,QRect crop={}) {
 const auto image=(crop.isNull()?w->grab():w->grab(crop)).toImage().convertToFormat(QImage::Format_RGB32);
 QHash<QRgb,int> n;QRgb best=0;int maximum=0;
 for(int y=2;y<image.height()-2;++y)for(int x=2;x<image.width()-2;++x){const auto p=image.pixel(x,y);const int count=++n[p];if(count>maximum){maximum=count;best=p;}}
 return QColor(best);
}
static bool isDark(QColor c){return c.lightness()<128;}
struct Page { QMainWindow *window; PdfWindow *model=nullptr; DrawingWindow *drawing=nullptr; FunctionDialogPanel *panel=nullptr; QDoubleSpinBox *field=nullptr; QCheckBox *box=nullptr; QString name; };
class ViewportInteractionTest {
public:
 static void run(bool only2d){
  DocumentTabsWindow host;host.resize(1320,900);host.show();
  QTemporaryDir files; if(!files.isValid())throw std::runtime_error("temp failed");
  QVector<Page> pages;
  const auto addPanel=[&](Page &p){p.panel=new FunctionDialogPanel(p.window,false);p.panel->setSizeIdentity("theme-tabs-probe-"+p.name);p.panel->setPanelOpacity(100);auto *form=p.panel->createScrollableForm();p.field=new QDoubleSpinBox(p.panel);p.field->setValue(12.5);form->addRow("Valore",p.field);p.box=new QCheckBox("Controllo",p.panel);p.box->setFocusPolicy(Qt::NoFocus);form->addRow(p.box);p.panel->show();};
  QString source;
  if(!only2d){
   for(int i=0;i<2;++i){auto *m=host.newModel();Page p{m,m,nullptr,nullptr,nullptr,nullptr,QString("model%1").arg(i)};ok(m->viewport_->createPrimitive(PrimitiveParameters{},"Test body"));source=files.filePath(QString("model%1.prt").arg(i));check(m->saveDocumentPath(source),"save isolated model");addPanel(p);pages.append(p);}
  }else{
   DocumentState empty;ExtrusionObject box;box.feature=BodyFeature::Primitive;empty.extrusions={box};CadViewport builder;builder.loadDocument(empty);source=files.filePath("empty.prt");ok(saveDocumentFile(source,builder.currentDocument()));
  }
  for(int i=0;i<2;++i){auto *d=new DrawingWindow(host.documentTabs());ok(d->createFromModel(source));host.addDocument(d,true);Page p{d,nullptr,d,nullptr,nullptr,nullptr,QString("drawing%1").arg(i)};addPanel(p);pages.append(p);}
  // Preserve explicit user choices independently from the normal theme surfaces.
  const QColor custom("#42662b");PdfWindow *customOwner=nullptr;BackgroundSettings customBackground;
  for(auto &p:pages)if(p.model){customOwner=p.model;customOwner->applyTreeBackground(custom);customBackground=customOwner->viewport_->background();customBackground.startColor=QColor("#214365");customBackground.endColor=QColor("#654321");customOwner->viewport_->setBackground(customBackground);break;}
  QMap<QString,QImage> icons;
  const auto verify=[&](bool dark,const QString &phase){
   check(isDark(QApplication::palette().color(QPalette::Window))==dark,phase+" application palette");
   for(auto &p:pages){
    // Palette check before reactivation exercises hidden pages as well.
    check(isDark(p.window->palette().color(QPalette::Window))==dark,phase+" "+p.name+" hidden/window palette");
    if(host.documentTabs()->indexOf(p.window)>=0)host.activateDocument(p.window);else p.window->show();
    p.panel->show();settle();
    const QString prefix=phase+" "+p.name;
    auto bars=p.window->findChildren<QToolBar *>();for(auto *bar:bars)bar->show();settle();int barChecks=0;
    for(auto *bar:bars)if(bar->isVisible()&&bar->width()>40&&bar->height()>10){
      const auto color=dominant(bar);check(isDark(color)==dark,prefix+" toolbar raster "+bar->objectName()+"="+color.name());
      check(isDark(bar->palette().color(QPalette::Button))==dark,prefix+" toolbar palette "+bar->objectName());++barChecks;
    }check(barChecks>0,prefix+" toolbar present");
    check(isDark(p.field->palette().color(QPalette::Base))==dark,prefix+" field palette");
    const auto fieldColor=dominant(p.field);check(isDark(fieldColor)==dark,prefix+" field raster="+fieldColor.name());
    check(isDark(p.panel->panelColor())==dark,prefix+" panel color");
    const auto panelColor=dominant(p.panel);check(isDark(panelColor)==dark,prefix+" panel raster="+panelColor.name());
    p.box->setChecked(false);const auto off=p.box->grab().toImage();p.box->setChecked(true);const auto on=p.box->grab().toImage();int contrast=0;
    for(int y=0;y<off.height();++y)for(int x=0;x<off.width();++x){const auto a=off.pixelColor(x,y),b=on.pixelColor(x,y);if(qMax(qMax(qAbs(a.red()-b.red()),qAbs(a.green()-b.green())),qAbs(a.blue()-b.blue()))>100)++contrast;}
    check(contrast>=6,prefix+" checkbox raster contrast");
    if(p.model){auto *tree=p.model->modelTree_;const auto color=dominant(tree->viewport());if(p.model==customOwner){check(tree->palette().color(QPalette::Base)==custom,prefix+" user tree palette retained");check(color==custom,prefix+" user tree raster retained");check(p.model->viewport_->background().startColor==customBackground.startColor && p.model->viewport_->background().endColor==customBackground.endColor,prefix+" canvas background retained");}else{check(isDark(tree->palette().color(QPalette::Base))==dark,prefix+" storyboard palette");check(isDark(color)==dark,prefix+" storyboard raster="+color.name());}}
    if(p.drawing){
      QByteArray bytes;ok(encodeDrawingSheet(p.drawing->editor()->sheet(),bytes));
      const auto previous=p.window->property("themeProbeDefinition").toByteArray();
      if(previous.isEmpty())p.window->setProperty("themeProbeDefinition",bytes);else check(previous==bytes,prefix+" drawing definition retained");
      p.drawing->editor()->fitSheet();settle();auto *canvas=p.drawing->editor()->canvas();
      const auto point=canvas->mapFromScene(QPointF(20,20));const auto capture=canvas->viewport()->grab().toImage();
      const auto sample=(QPointF(point)*capture.devicePixelRatio()).toPoint();
      check(capture.rect().contains(sample)&&capture.pixelColor(sample)==QColor(Qt::white),prefix+" paper raster stays white");
      // Technical ink belongs to the drawing, independent from the interface theme.
      const auto size=p.drawing->editor()->sheet().size;
      QImage sceneImage(1000,700,QImage::Format_RGB32);sceneImage.fill(Qt::white);QPainter scenePainter(&sceneImage);
      canvas->scene()->render(&scenePainter,QRectF(0,0,1000,700),QRectF(QPointF(0,0),size),Qt::IgnoreAspectRatio);scenePainter.end();
      int sceneInk=0;for(int y=0;y<sceneImage.height();++y)for(int x=0;x<sceneImage.width();++x)sceneInk+=sceneImage.pixelColor(x,y).lightness()<128;
      check(sceneInk>100,prefix+" technical border/title text raster present");
      const auto imageBytes=QByteArray(reinterpret_cast<const char *>(sceneImage.constBits()),sceneImage.sizeInBytes());
      const auto previousImage=p.window->property("themeProbePaperRaster").toByteArray();
      if(previousImage.isEmpty())p.window->setProperty("themeProbePaperRaster",imageBytes);else check(previousImage==imageBytes,prefix+" technical sheet raster identical across themes");
      // A frame can be queued after Qt's palette propagation. Wait for that frame before sampling the visible canvas.
      const auto canvasInk=[&]{const auto paperRect=canvas->mapFromScene(QRectF(QPointF(0,0),size)).boundingRect().intersected(canvas->viewport()->rect()).adjusted(2,2,-2,-2);const auto actual=canvas->viewport()->grab(paperRect).toImage();int ink=0;for(int y=0;y<actual.height();++y)for(int x=0;x<actual.width();++x)ink+=actual.pixelColor(x,y).lightness()<128;return ink;};
      QElapsedTimer frame;frame.start();int actualInk=canvasInk();while(actualInk<100 && frame.elapsed()<1000){QApplication::processEvents(QEventLoop::AllEvents,20);QThread::msleep(5);actualInk=canvasInk();}
      check(actualInk>=100,prefix+" visible technical border/title text raster present");

    }
    auto *menu=p.window->findChild<QMenu *>("applicationThemeMenu");check(menu&&menu->actions().size()==3,prefix+" theme menu present");
    if(menu){menu->popup({50,50});settle();const auto image=menu->grab().toImage();const auto point=QPointF(menu->width()/2.0,menu->actionGeometry(menu->actions().back()).bottom()+3)*image.devicePixelRatio();const auto color=image.pixelColor(point.toPoint());check(color.alpha()>200,prefix+" menu surface opaque interior");check(isDark(color)==dark,prefix+" menu raster="+color.name());check(isDark(menu->palette().color(QPalette::WindowText))!=dark,prefix+" menu palette");int selected=0;for(auto *a:menu->actions())selected+=a->isChecked();check(selected==1,prefix+" menu single selection");menu->hide();}
    // Render the actual action icon, whose engine predates the change.
    QAction *iconAction=p.window->findChild<QAction *>(p.model?"documentSaveAction":"drawingSaveAction");check(iconAction&&!iconAction->icon().isNull(),prefix+" actual icon present");
    if(iconAction){QImage image(32,32,QImage::Format_ARGB32);image.fill(Qt::transparent);QPainter painter(&image);iconAction->icon().paint(&painter,QRect(0,0,32,32));painter.end();int visible=0;for(int y=0;y<32;++y)for(int x=0;x<32;++x)if(image.pixelColor(x,y).alpha()>128)++visible;check(visible>30,prefix+" icon raster visible");const QString key=p.name+QString(dark?"-dark":"-light");if(icons.contains(key))check(image==icons[key],prefix+" icon raster stable after reactivation");else icons[key]=image;const QString other=p.name+QString(dark?"-light":"-dark");if(icons.contains(other))check(image!=icons[other],prefix+" icon raster follows theme");image.save(output+"/"+phase+"-"+p.name+"-icon.png");}
    p.window->grab().save(output+"/"+phase+"-"+p.name+".png");p.panel->grab().save(output+"/"+phase+"-"+p.name+"-panel.png");
   }
  };
  // Drive the actual menus, alternating a 3D and 2D origin where possible.
  const auto choose=[&](int index){auto *menu=pages[index==2?0:(only2d?0:2)].window->findChild<QMenu *>("applicationThemeMenu");check(menu,"origin menu exists");if(menu)menu->actions()[index]->trigger();else setApplicationThemeMode(*qApp,index==2?ApplicationThemeMode::Dark:index==1?ApplicationThemeMode::Light:ApplicationThemeMode::Desktop);settle();};
  choose(1);verify(false,"light0");choose(2);verify(true,"dark1");choose(1);verify(false,"light2");choose(2);verify(true,"dark3");
  if(!only2d){auto *m=new PdfWindow;Page p{m,m,nullptr,nullptr,nullptr,nullptr,"detached-model"};m->show();addPanel(p);pages.append(p);}
  auto *d=new DrawingWindow;ok(d->createFromModel(source));d->resize(host.size());d->show();Page dp{d,nullptr,d,nullptr,nullptr,nullptr,"detached-drawing"};addPanel(dp);pages.append(dp);
  // New documents must inherit the current global style without a palette event.
  if(!only2d){auto *m=host.newModel();Page p{m,m,nullptr,nullptr,nullptr,nullptr,"new-model-dark"};addPanel(p);pages.append(p);}
  {auto *d=new DrawingWindow(host.documentTabs());ok(d->createFromModel(source));host.addDocument(d,true);Page p{d,nullptr,d,nullptr,nullptr,nullptr,"new-drawing-dark"};addPanel(p);pages.append(p);}
  verify(true,"dark-new");
  choose(0);
#if QT_VERSION >= QT_VERSION_CHECK(6,5,0)
  for(bool dark:{false,true,false}){check(QMetaObject::invokeMethod(qApp->styleHints(),"colorSchemeChanged",Qt::DirectConnection,Q_ARG(Qt::ColorScheme,dark?Qt::ColorScheme::Dark:Qt::ColorScheme::Light)),"desktop scheme signal");settle();verify(dark,dark?"desktop-dark":"desktop-light");}
#endif
  // Changing sheet rebuilds view menus; the same theme entry must survive exactly once.
  for(auto &p:pages)if(p.drawing){ok(p.drawing->addSheet("Secondo foglio",{420,297}));ok(p.drawing->setActiveSheet(0));p.window->setProperty("themeProbeDefinition",QByteArray());p.window->setProperty("themeProbePaperRaster",QByteArray());check(p.window->findChildren<QMenu *>("applicationThemeMenu").size()==1,"exactly one 2D theme menu after sheet rebuild");}
  choose(2);verify(true,"sheets-dark");
  // Close the final 3D page: the remaining 2D session still owns theme switching.
  for(int i=pages.size()-1;i>=0;--i)if(pages[i].model){auto p=pages.takeAt(i);delete p.panel;const int index=host.documentTabs()->indexOf(p.window);if(index>=0)host.documentTabs()->removeTab(index);delete p.window;}
  choose(1);verify(false,"only2d-light");choose(2);verify(true,"only2d-dark");
  for(auto &p:pages){p.panel->hide();delete p.panel;p.panel=nullptr;}
  // Clean destruction without saving or changing any original CAD.
  for(auto &p:pages){const int index=host.documentTabs()->indexOf(p.window);if(index>=0)host.documentTabs()->removeTab(index);delete p.window;p.window=nullptr;}
 }
};
int main(int argc,char **argv){QApplication app(argc,argv);QCoreApplication::setOrganizationName("ForgeCAD-ThemeTabs-Isolated");QCoreApplication::setApplicationName("Probe");QTemporaryDir configuration;QSettings::setDefaultFormat(QSettings::IniFormat);QSettings::setPath(QSettings::IniFormat,QSettings::UserScope,configuration.path());qputenv("FORGECAD_RECOVERY_DIR",(configuration.path()+"/recovery").toUtf8());output=app.arguments().value(1,"/tmp/theme-tabs-review/tests/captures");QDir().mkpath(output);try{initializeApplicationTheme(app);ViewportInteractionTest::run(app.arguments().contains("--only-2d"));}catch(const std::exception &e){std::cerr<<e.what()<<std::endl;return 2;}std::cout<<"RESULT checks="<<checks<<" failures="<<failures<<" backend="<<QGuiApplication::platformName().toStdString()<<" Qt="<<qVersion()<<std::endl;return failures?1:0;}
