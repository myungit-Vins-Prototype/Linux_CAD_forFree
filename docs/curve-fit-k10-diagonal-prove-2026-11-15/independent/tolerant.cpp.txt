CurvePtr<3> profile(bool rational,double scale,double start=.1){std::vector<double>k(7,0);k.insert(k.end(),6,start);k.insert(k.end(),6,start+.001);k.insert(k.end(),7,1);std::vector<Vec3>p;std::vector<double>w;for(int i=0;i<19;++i){double x=0;for(int j=1;j<=6;++j)x+=k[i+j]/6;p.emplace_back(scale*x,0,i==9?.0032:0);if(rational)w.push_back(1+.1*i/18.);}return std::make_shared<BSplineCurve<3>>(6,k,p,w);}
Body sheet(Frame3 f,double width,double height){std::vector<Vec3>v;for(auto q:std::vector<Vec3>{{0,0,0},{width,0,0},{width,height,0},{0,height,0}})v.push_back(f.toGlobal(q));std::vector<Body::BuildEdge>e;for(int i=0;i<4;++i){auto a=v[i],b=v[(i+1)%4];e.push_back({i,(i+1)%4,std::make_shared<Line<3>>(a,b-a),{0,distance(a,b)},0});}Body::BuildFace face;face.surface=std::make_shared<Plane>(f);face.loops={{{0,true,nullptr,0},{1,true,nullptr,0},{2,true,nullptr,0},{3,true,nullptr,0}}};auto result=Body::buildSheet(v,e,{face});if(computePCurves(result))throw std::runtime_error("source pcurve");return result;}
int main(){std::cout<<std::setprecision(17);for(int kind=0;kind<4;++kind){auto source=sheet(Frame3(),1,1);if(kind==0||kind==3){auto v=source.vertices().front();source.vertex(v).point[0]+=kind==3?1.9e-8:1e-5;source.vertex(v).tolerance=2e-5;}if(kind==1){for(auto id:source.fins()){auto&f=source.fin(id);const auto&e=source.edge(f.edge);auto a=f.pcurve->point(e.range.lo)+Vec2(1e-5,0),b=f.pcurve->point(e.range.hi)+Vec2(1e-5,0);f.pcurve=std::make_shared<BSplineCurve<2>>(1,std::vector<double>{e.range.lo,e.range.lo,e.range.hi,e.range.hi},std::vector<Vec2>{a,b});f.pcurveTolerance=2e-5;}}auto target=std::make_shared<ExtrusionSurface>(profile(false,1),Vec3(0,1,0));SurfaceWrapOptions o;o.tolerance=1e-6;double dev=123;BodyHistory history(1,{1},{4},{4},2,3,4);std::cout<<"case="<<kind<<" source_issues="<<checkBody(source,{true,true}).size();auto token=std::make_shared<std::atomic<bool>>(kind==2);try{CancellationScope cancel(token);auto result=wrapBodyOnSurface(source,*target,o,&history,&dev);auto v=source.vertices().front();double vertexError=distance(result.vertex(v).point,target->point(source.vertex(v).point.x(),source.vertex(v).point.y()));std::cout<<" accepted=1 deviation="<<dev<<" vertex_error="<<vertexError<<" history_faces="<<history.resultSize(EntityKind::Face)<<" issues="<<checkBody(result,{true,true}).size()<<"\n";}catch(const OperationCancelled&){std::cout<<" cancelled=1 deviation="<<dev<<" history_faces="<<history.resultSize(EntityKind::Face)<<"\n";}catch(const std::exception&e){std::cout<<" accepted=0 deviation="<<dev<<" history_faces="<<history.resultSize(EntityKind::Face)<<" reason="<<e.what()<<"\n";}}}
