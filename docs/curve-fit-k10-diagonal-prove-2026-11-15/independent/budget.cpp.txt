CurvePtr<3> profile(bool rational,double scale,double start=.1){std::vector<double>k(7,0);k.insert(k.end(),6,start);k.insert(k.end(),6,start+.001);k.insert(k.end(),7,1);std::vector<Vec3>p;std::vector<double>w;for(int i=0;i<19;++i){double x=0;for(int j=1;j<=6;++j)x+=k[i+j]/6;p.emplace_back(scale*x,0,i==9?.0032:0);if(rational)w.push_back(1+.1*i/18.);}return std::make_shared<BSplineCurve<3>>(6,k,p,w);}
