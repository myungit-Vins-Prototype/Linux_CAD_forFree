import json,re

 txt=(p/('kernel-'+result['filter']+'.log')).read_text();names=re.findall(r'^\[  OK  \] (.+)$',txt,re.M);failed=re.findall(r'^\[ FAIL \] (.+)$',txt,re.M);assert result['exit']==0 and names and not failed,result
 executions+=len(names)
 for n in names:seen.setdefault(n,[]).append(result['filter'])

r=dict(kernel_distinct=len(seen),kernel_executions=executions,kernel_seconds=sum(x['seconds'] for x in results),kernel_cases=seen,ctest_suites=len(c['tests']),ctest_seconds=c['seconds']);(p/'test-summary.json').write_text(json.dumps(r,indent=2));print({k:v for k,v in r.items() if k!='kernel_cases'})
