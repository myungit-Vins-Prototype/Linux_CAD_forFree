#!/usr/bin/env python3
# Link con oggetti APP correnti, senza modificare build o test TU generale.
# Uso: python3 ui_probe.py.txt /repo [/tmp/probe-dir] [--diagnostic]
import json,pathlib,shlex,subprocess,sys,os
root=pathlib.Path(sys.argv[1]).resolve();diagnostic='--diagnostic' in sys.argv[2:]
paths=[a for a in sys.argv[2:] if a!='--diagnostic']
dest=pathlib.Path(paths[0] if paths else '/tmp/ellipse-full-ui-private-probe');dest.mkdir(exist_ok=True)
fixture='ui_diagnostic.cpp.txt' if diagnostic else 'full_ui_probe.cpp.txt'
(dest/'probe.cpp').write_text(pathlib.Path(__file__).with_name(fixture).read_text())
(dest/'ellipse_extremes_fixture.h').write_text((root/'tests/test_ellipse_extremes.h').read_text())
entries=json.loads((root/'forgecad-release/compile_commands.json').read_text())
args=shlex.split(next(x for x in entries if x['file'].endswith('/tests/test_viewport.cpp'))['command'])
i=args.index('-o');del args[i:i+2];args=args[:args.index('-c')]
args=[x for x in args if x!='-O3']+['-O0']
subprocess.run(args+['-c',str(dest/'probe.cpp'),'-o',str(dest/'probe.o')],check=True)
link=shlex.split((root/'forgecad-release/CMakeFiles/forgecad.dir/link.txt').read_text())
link=[x for x in link if not x.startswith('-Wl,--dependency-file=') and not x.endswith('/forgeCad2026.cpp.o')]
i=link.index('-o');link[i+1]=str(dest/'probe');link.insert(1,str(dest/'probe.o'))
subprocess.run(link,cwd=root/'forgecad-release',check=True)
env=os.environ.copy();env.update(QT_QPA_PLATFORM='offscreen',QT_QPA_PLATFORMTHEME='',QT_QPA_PLATFORMTHEME_QT6='',QT_STYLE_OVERRIDE='Fusion',QT_QPA_OFFSCREEN_NO_GLX='1',LIBGL_ALWAYS_SOFTWARE='1')
subprocess.run([str(dest/'probe')],env=env,check=True)
