from pathlib import Path
import subprocess,time,json,sys,hashlib,xml.etree.ElementTree as ET,re
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/k10-instance-carrier-review/root')
targets=['forgekernel_surface_curve_bounds_tests','forgekernel_mapping_nonplane_validation_tests','forgekernel_surface_mapping_tests','forgekernel_sheet_join_validation_tests','forgekernel_surface_instances_tests','forgekernel_tolerance_tests']
def run(name,args,timeout=900):
 start=time.monotonic()
 with (T/(name+'.log')).open('w') as out:
  try:code=subprocess.run(args,cwd=R,stdout=out,stderr=subprocess.STDOUT,timeout=timeout).returncode
  except subprocess.TimeoutExpired:code=124
 j={'command':args,'exit':code,'seconds':time.monotonic()-start};(T/(name+'-result.json')).write_text(json.dumps(j,indent=2)+'\n');print(name,j,flush=True)
 if code:print((T/(name+'.log')).read_text()[-5000:]);raise SystemExit(code)
 return j
if sys.argv[1]=='build':
 j=run(sys.argv[2] if len(sys.argv)>2 else 'build',['cmake','--build','forgecad-release','--target','forgecad']+targets+['--parallel','2']);bins=['forgecad-release/forgecad','forgecad-release/kernel/libforgekernel.a']+['forgecad-release/kernel/'+t for t in targets];(T/'binaries.json').write_text(json.dumps({f:hashlib.sha256((R/f).read_bytes()).hexdigest() for f in bins},indent=2)+'\n')
elif sys.argv[1]=='tests':
 actual={t['name'] for t in json.loads(subprocess.check_output(['ctest','--test-dir','forgecad-release','--show-only=json-v1'],cwd=R))['tests']};assert set(targets)<=actual,set(targets)-actual
 j=run('ctest',['ctest','--test-dir','forgecad-release','--output-on-failure','-V','--parallel','1','--output-junit',str(T/'ctest.xml'),'-R','^('+'|'.join(map(re.escape,targets))+')$']);cases=list(ET.parse(T/'ctest.xml').getroot().iter('testcase'));assert len(cases)==len(targets);assert not any(c.find('skipped') is not None or c.find('failure') is not None for c in cases);j.update(actual_passed=len(cases),skipped=[]);(T/'ctest-result.json').write_text(json.dumps(j,indent=2)+'\n')
elif sys.argv[1]=='architecture':run('architecture-generate',['python3','tools/generate_architecture.py','--build-dir','forgecad-release','--jobs','4'])
