void require(bool v,const char*m) {if(!v)throw std::runtime_error(m);}
std::string number(double d) {
    std::ostringstream o;
    o << '"' << std::hexfloat << d << '"';
    return o.str();
}
template <class T> void array(std::ostream &o, const T &v) {
    o << '[';
    bool sep = false;
    for (auto x : v) {
        if (sep)
            o << ',';
        sep = true;
        o << number(x);
    }
    o << ']';
}
template <int N> void vector(std::ostream &o, const Vec<N> &v) {
    o << '[';
    for (int i = 0; i < N; ++i) {
        if (i)
            o << ',';
        o << number(v[i]);
    }
    o << ']';
}
template <int N> void curveJson(std::ostream &o, const CurvePtr<N> &c) {
    if (auto s = std::dynamic_pointer_cast<const BSplineCurve<N>>(c)) {
        o << "{\"type\":\"bspline\",\"degree\":" << s->degree()
          << ",\"knots\":";
        array(o, s->knots());
        o << ",\"poles\":[";
        for (int i = 0; i < s->poleCount(); ++i) {
            if (i)
                o << ',';
            vector(o, s->poles()[i]);
        }
        o << "],\"weights\":[";
        for (int i = 0; i < s->poleCount(); ++i) {
            if (i)
                o << ',';
            o << number(s->weight(i));
        }
        o << "]}";
    } else if (auto l = std::dynamic_pointer_cast<const Line<N>>(c)) {
        o << "{\"type\":\"line\",\"origin\":";
        vector(o, l->origin());
        o << ",\"direction\":";
        vector(o, l->direction());
        o << '}';
    } else if (auto t = std::dynamic_pointer_cast<const TrimmedCurve<N>>(c)) {
        o << "{\"type\":\"trimmed\",\"basis\":";
        curveJson(o, t->basis());
        o << '}';
    } else
        throw std::runtime_error("test export: unsupported curve wrapper");
}
void surfaceJson(std::ostream &o, const BSplineSurface &s, int depth = 0) {
    if (depth > 32) throw std::runtime_error("export chain exceeds bounded input");
    if (s.basis()) {
        o << "{\"type\":\"instance\",\"placement\":{\"matrix\":[";
        for(int i=0;i<3;++i) {if(i)o<<',';o<<'[';for(int j=0;j<3;++j) {if(j)o<<',';o<<number(s.placement().matrix(i,j));}o<<']';}
        o << "],\"translation\":";vector(o,s.placement().translationPart());o<<"},\"basis\":";
        surfaceJson(o,*s.basis(),depth+1);o<<'}';return;
    }
    o << "{\"uDegree\":" << s.uDegree() << ",\"vDegree\":" << s.vDegree()
      << ",\"uKnots\":";
    array(o, s.uKnots());
    o << ",\"vKnots\":";
    array(o, s.vKnots());
    o << ",\"uPoleCount\":" << s.uPoleCount()
      << ",\"vPoleCount\":" << s.vPoleCount() << ",\"poles\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            if (i || j)
                o << ',';
            vector(o, s.pole(i, j));
        }
    o << "],\"weights\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            if (i || j)
                o << ',';
            o << number(s.weight(i, j));
        }
    o << "]}";
}

Body rationalTarget() {
 std::vector<double> knots{0,0,0,1,1,1},weights;std::vector<Vec3> poles;
 const double x[]{0,.25,1},w[]{1,2,3};
 for(int i=0;i<3;++i)for(int j=0;j<3;++j){poles.push_back({x[i],.5*j,0});weights.push_back(w[i]);}
 auto s=std::make_shared<BSplineSurface>(2,2,knots,knots,3,3,poles,weights);
 std::vector<Vec3> points{s->point(0,0),s->point(1,0),s->point(1,1),s->point(0,1)};
 std::vector<Body::BuildEdge> edges{{0,1,s->vIso(0),{0,1},0},{1,2,s->uIso(1),{0,1},0},{3,2,s->vIso(1),{0,1},0},{0,3,s->uIso(0),{0,1},0}};
 auto pc=[](Vec2 a,Vec2 d){return std::make_shared<Line<2>>(a,d);};
 Body::BuildFace f;f.surface=s;f.loops={{{0,true,pc({0,0},{1,0}),0},{1,true,pc({1,0},{0,1}),0},{2,false,pc({0,1},{1,0}),0},{3,false,pc({0,0},{0,1}),0}}};
 return Body::buildSheet(points,edges,{f});
}


int main(int argc,char**argv) {
 try {
  require(argc==2,"output.json required");
  const Body source=makePlaneSheet(Frame3(),.1),originalTarget=rationalTarget();
  const auto pose=Transform3::translation({.2,-.3,.4})*Transform3::rotation({0,0,0},{.2,1,.3},.25)*Transform3::scaling({0,0,0},1.25);
  const Body target=transformBody(originalTarget,pose);const auto carrier=std::dynamic_pointer_cast<const BSplineSurface>(target.face(target.faces().front()).surface);
  require(carrier&&carrier->basis(),"fixture must use real stored instance");
  const auto sourceBefore=writeBodyBinary(source),targetBefore=writeBodyBinary(target),originalBefore=writeBodyBinary(originalTarget);
  SurfaceWrapOptions options;options.anchor={.5,.5};const double cap=.8*std::min(options.tolerance,linearResolution());
  const auto build=[&](BodyHistory*h,double*d){SurfaceQuiltMap quilt({{&target,target.faces().front()}},options.anchor);return quilt.bend(source,options,h,d);};
  BodyHistory history;double deviation=-1;const Body output=build(&history,&deviation);
  require(checkBody(output,{true,true}).empty(),"invalid actual output");
  double noHistoryDeviation=-1;const Body noHistory=build(nullptr,&noHistoryDeviation);
  require(writeBodyBinary(noHistory)==writeBodyBinary(output)&&noHistoryDeviation==deviation,"historynull changes actual output");
  require(writeBodyBinary(source)==sourceBefore&&writeBodyBinary(target)==targetBefore&&writeBodyBinary(originalTarget)==originalBefore,"input mutation");
  require(history.operandCount()==1,"bad operand count");
  std::ofstream out(argv[1]);out<<"{\"scope\":\"public-instance-final-fins\",\"prefitCap\":"<<number(cap)<<",\"sourceImmutable\":true,\"targetImmutable\":true,\"originalTargetImmutable\":true,\"historyNullByteIdentical\":true,\"deviation\":"<<number(deviation)<<",\"records\":[";
  int count=0;
  for(auto face:output.faces()) {
   const auto&fo=history.origins(EntityKind::Face,face.index);require(history.isDeclared(EntityKind::Face,face.index)&&fo.size()==1&&fo[0].operand==0&&fo[0].kind==EntityKind::Face&&fo[0].relation==HistoryRelation::Modified&&source.contains(FaceId(fo[0].index)),"invalid face origin");
   const FaceId sourceFace(fo[0].index);require(output.face(face).surface==carrier,"carrier instance not preserved");
   for(auto loop:output.face(face).loops)for(auto id:output.loopFins(loop)) {
    const auto&fin=output.fin(id);const auto&e=output.edge(fin.edge);const auto&eo=history.origins(EntityKind::Edge,fin.edge.index);
    require(history.isDeclared(EntityKind::Edge,fin.edge.index)&&eo.size()==1&&eo[0].operand==0&&eo[0].kind==EntityKind::Edge&&eo[0].relation==HistoryRelation::Modified&&source.contains(EdgeId(eo[0].index)),"invalid edge origin");
    const EdgeId sourceEdge(eo[0].index);const auto&original=source.edge(sourceEdge);require(e.range.lo==original.range.lo&&e.range.hi==original.range.hi,"source parameter changed");
    int sourceFin=-1;for(auto l:source.face(sourceFace).loops)for(auto f:source.loopFins(l))if(source.fin(f).edge==sourceEdge&&source.fin(f).sense==fin.sense){require(sourceFin<0,"ambiguous source incidence");sourceFin=f.index;}require(sourceFin>=0,"fin sense or ancestry changed");
    const auto proof=detail::surfaceCurveResidualBound(e.curve,carrier,fin.pcurve,e.range,cap);require(proof.status==detail::SurfaceCurveBoundStatus::Certified&&std::isfinite(proof.upperBound)&&proof.upperBound>=0&&proof.upperBound<=cap,"instance final not Certified");require(e.tolerance==0&&fin.pcurveTolerance==cap,"fixed budget metadata changed");
    if(count++)out<<',';
    out<<"{\"label\":\"public-instance\",\"face\":"<<face.index<<",\"edge\":"<<fin.edge.index<<",\"fin\":"<<id.index<<",\"sense\":"<<fin.sense<<",\"sourceFace\":"<<sourceFace.index<<",\"sourceEdge\":"<<sourceEdge.index<<",\"sourceFin\":"<<sourceFin<<",\"budget\":"<<number(cap)<<",\"edgeTolerance\":"<<number(e.tolerance)<<",\"pcurveTolerance\":"<<number(fin.pcurveTolerance)<<",\"range\":["<<number(e.range.lo)<<','<<number(e.range.hi)<<"],\"runtimeStatus\":\"Certified\",\"runtimeUpper\":"<<number(proof.upperBound)<<",\"curve\":";curveJson(out,e.curve);out<<",\"pcurve\":";curveJson(out,fin.pcurve);out<<",\"surface\":";surfaceJson(out,*carrier);out<<'}';
   }
  }
  out<<"]}\n";require(count==4&&bool(out),"wrong final count or write failed");std::cout<<"Public actual stored instance4/4Certified cap="<<cap<<" noHistory byte-identical, inputs immutable\n";
 } catch(const std::exception&e){std::cerr<<e.what()<<'\n';return 1;}
}
