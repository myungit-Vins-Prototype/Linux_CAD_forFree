#!/usr/bin/env python3
"""Run ONLY after root gives a stable final archive path and SHA.
No archive copy is made. Keep generated files in private staging, outside sealed proof.
"""
from pathlib import Path
import sys,json,hashlib,time,subprocess
root=Path(__file__).parent;repo=Path('/home/myungit/Linux_CAD_forFree');library=Path(sys.argv[1]);expected=sys.argv[2]
def digest(p):return hashlib.sha256(p.read_bytes()).hexdigest()
assert digest(library)==expected,'Library not ready or changed'
meta=json.loads((root/'compile-public-instance.json').read_text());assert meta['exit']==0 and digest(root/'export_public_instance.cpp.txt')==meta['sourceSHA256']
cmd=['g++',str(root/'export-public-instance.o'),str(library),'-lpthread','-o',str(root/'export-public-instance')];start=time.monotonic();r=subprocess.run(cmd,cwd=repo,capture_output=True,text=True)
(root/'final-link.log').write_text(r.stdout+r.stderr);(root/'final-link.json').write_text(json.dumps({'argv':cmd,'cwd':str(repo),'exit':r.returncode,'elapsed':time.monotonic()-start,'librarySHA256':expected,'libraryBytes':library.stat().st_size,'objectSHA256':digest(root/'export-public-instance.o')},indent=2)+'\n');assert r.returncode==0,r.stderr;assert digest(library)==expected
output=root/'actual-public-instance-final.json';cmd=[str(root/'export-public-instance'),str(output)];start=time.monotonic();r=subprocess.run(cmd,cwd=repo,capture_output=True,text=True);(root/'final-public-instance.log').write_text(r.stdout+r.stderr);(root/'final-public-run.json').write_text(json.dumps({'argv':cmd,'exit':r.returncode,'elapsed':time.monotonic()-start,'outputSHA256':digest(output) if output.exists() else None},indent=2)+'\n');print(r.stdout+r.stderr,flush=True);assert r.returncode==0
cmd=[sys.executable,str(root/'verify_public_fins.py.txt'),str(output),str(root/'final-instance-fraction-result.json')];start=time.monotonic();r=subprocess.run(cmd,cwd=repo,capture_output=True,text=True);(root/'final-instance-fraction.log').write_text(r.stdout+r.stderr);(root/'final-fraction-run.json').write_text(json.dumps({'argv':cmd,'exit':r.returncode,'elapsed':time.monotonic()-start},indent=2)+'\n');assert r.returncode==0,r.stderr
result=json.loads((root/'final-instance-fraction-result.json').read_text());assert result['pass'] and len(result['records'])==4
print('Public API4actual instance fins + historynull + independent Fraction4 PASS',flush=True)
