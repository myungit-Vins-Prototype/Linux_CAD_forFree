namespace {
void checkOffset(const CurvePtr<3> &curve, bool revolution = false) {
    SurfacePtr surface = revolution ? SurfacePtr(std::make_shared<RevolutionSurface>(curve, Vec3(), Vec3(0, 0, 1)))
                                    : SurfacePtr(std::make_shared<ExtrusionSurface>(curve, Vec3(0, 1, 0)));
    Interval range = curve->domain();
    auto f = [&](double t) {
        return revolution ? surface->point(0, t) + .001 * surface->normal(0, t)
                          : surface->point(t, 0) + .001 * surface->normal(t, 0);
    };
    auto bound = revolution ? detail::revolutionOffsetErrorBound(curve, range, .001, {}, Vec3(0, 0, 1))
                            : detail::extrusionOffsetErrorBound(curve, range, .001, Vec3(0, 1, 0));
    FK_CHECK(bool(bound));
    double deviation = -1;
    auto result = fitCurveCertified(f, range, curve->breakpoints(range), 1e-7, bound, &deviation);
    FK_CHECK(deviation >= 0 && deviation <= 1e-7);
    double worst = 0;
    for (int i = 0; i <= 1000; ++i) {
        double t = range.lo + range.length() * i / 1000.;
        worst = std::max(worst, distance(result->point(t), f(t)));
    }
    FK_CHECK(worst <= deviation + 1e-12);
}
} // namespace
FK_TEST(CurveFitBoundsRationalScaleTrimTransform) {
    for (double scale : {1., 1e-100, 1e100}) {
        auto curve = std::make_shared<BSplineCurve<3>>(2, std::vector<double>{0, 0, 0, 1, 1, 1},
                                                       std::vector<Vec3>{{3, 0, 0}, {3, 0, 1}, {2, 0, 1}},
                                                       std::vector<double>{scale, scale * .7071067811865476, scale});
        checkOffset(curve);
        checkOffset(curve, true);
        auto trim = std::make_shared<TrimmedCurve<3>>(curve, .13, .87);
        checkOffset(trim);
        checkOffset(trim, true);
        auto moved = std::make_shared<TransformedCurve>(trim, Transform3::translation(Vec3(.1, .2, .3)) *
                                                                  Transform3::rotation({}, Vec3(0, 1, 0), .3));
        checkOffset(moved);
        checkOffset(moved, true);
    }
}
FK_TEST(CurveFitBoundsMultipleSpansAndUnclamped) {
    auto curve = std::make_shared<BSplineCurve<3>>(
        3, std::vector<double>{0, 1, 2, 3, 4, 5, 6, 7, 8},
        std::vector<Vec3>{{2, 0, 0}, {2.1, 0, .3}, {2.4, 0, .5}, {2.5, 0, .8}, {2.8, 0, 1}});
    checkOffset(curve);
    checkOffset(curve, true);
}
FK_TEST(CurveFitBoundsFailuresAreClosed) {
    const auto f = [](double t) { return Vec3(t, 0, 0); };
    double deviation = 123;
    FK_CHECK_THROWS(fitCurveCertified(f, {0, 1}, {}, 1e-7, {}, &deviation));
    FK_CHECK(deviation == 123);
    FK_CHECK_THROWS(fitCurveCertified(
        f, {0, 1}, {}, 1e-7,
        [](const Interval &, const std::array<Vec3, 4> &) { return std::numeric_limits<double>::infinity(); },
        &deviation));
    FK_CHECK(deviation == 123);
    FK_CHECK_THROWS(fitCurve(f, {0, 1}, {}, 0, &deviation));
    FK_CHECK(deviation == 123);
    FK_CHECK_THROWS(fitCurve([](double) { return Vec3(std::numeric_limits<double>::infinity(), 0, 0); }, {0, 1}, {},
                             1e-7, &deviation));
    FK_CHECK(deviation == 123);
    FK_CHECK_THROWS(fitCurve([](double t) { return Vec3(t < .5 ? 0 : 1, 0, 0); }, {0, 1}, {}, 1e-12, &deviation));
    FK_CHECK(deviation == 123);
    auto curve =
        std::make_shared<BSplineCurve<3>>(1, std::vector<double>{0, 0, 1, 1}, std::vector<Vec3>{{0, 0, 0}, {1, 0, 0}});
    auto bound = detail::extrusionOffsetErrorBound(curve, {0, 1}, .01, Vec3(0, 1, 0));
    FK_CHECK(!std::isfinite(bound({-1, 1}, {Vec3(), Vec3(), Vec3(), Vec3()})));
    auto token = std::make_shared<std::atomic<bool>>(true);
    CancellationScope scope(token);
    bool cancelled = false;
    try {
        detail::extrusionOffsetErrorBound(curve, {0, 1}, .01, Vec3(0, 1, 0));
    } catch (const OperationCancelled &) {
        cancelled = true;
    }
    FK_CHECK(cancelled);
}
