static Vec3 rotate(Vec3 p){const Vec3 a=normalized(Vec3(1,2,3));const double c=std::cos(.71),s=std::sin(.71);return c*p+s*cross(a,p)+(1-c)*dot(a,p)*a;}
static std::shared_ptr<BSplineCurve<3>> source(double scale,bool pose,bool rational){std::vector<Vec3> p;std::vector<double>w;for(int i=0;i<=27;++i){Vec3 q=scale*Vec3(i/27.,0,z[i]);if(pose)q=rotate(q)+scale*Vec3(11,-7,3);p.push_back(q);if(rational)w.push_back(1+.1*std::sin(i*.4));}std::vector<double>k(28,0);k.insert(k.end(),28,1);return std::make_shared<BSplineCurve<3>>(27,k,p,w);}
