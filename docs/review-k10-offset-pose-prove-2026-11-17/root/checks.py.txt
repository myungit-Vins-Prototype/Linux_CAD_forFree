from pathlib import Path
import subprocess,json,time,sys,re,xml.etree.ElementTree as ET
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/k10-offset-pose-review/root')
def run(name,args,timeout=None):
 start=time.monotonic()
 with(T/(name+'.log')).open('w') as log:
  try:code=subprocess.run(args,cwd=R,stdout=log,stderr=subprocess.STDOUT,timeout=timeout).returncode
  except subprocess.TimeoutExpired:code=124
 j={'exit':code,'seconds':time.monotonic()-start,'command':args};(T/(name+'-result.json')).write_text(json.dumps(j,indent=2)+'\n');print(name,j,flush=True)
 if code:print((T/(name+'.log')).read_text()[-4000:],flush=True);raise SystemExit(code)
 return j
if __name__=='__main__':
 if sys.argv[1]=='build':run('build',['cmake','--build','forgecad-release','--parallel','2'])
 elif sys.argv[1]=='general':run('general',['./forgecad-release/kernel/forgekernel_tests'],timeout=1200)
 elif sys.argv[1]=='tests':
  wanted=['forgekernel_shell_pose_tests', 'forgekernel_offset_validation_tests', 'forgekernel_curve_fit_validation_tests', 'forgekernel_offset_pcurve_tests', 'forgekernel_offset_split_crease_tests', 'forgekernel_edge_offset_fit_validation_tests', 'forgekernel_surface_curve_fit_validation_tests', 'forgekernel_boundary_fit_validation_tests', 'forgekernel_sheet_join_validation_tests', 'forgekernel_surface_curve_bounds_tests', 'forgekernel_sheet_fit_validation_tests', 'forgekernel_blend_cavity_tests', 'forgekernel_blend_cavity_regression_tests', 'forgekernel_blend_contact_break_tests', 'forgekernel_history_tests', 'forgekernel_surface_tools_tests', 'forgekernel_surface_parameterization_tests', 'forgecad_surface_tools_tests', 'forgecad_extension_panel_tests', 'forgecad_all_history_tests', 'forgecad_blend_history_tests']
  actual={t['name'] for t in json.loads(subprocess.check_output(['ctest','--test-dir','forgecad-release','--show-only=json-v1'],cwd=R))['tests']};assert set(wanted)<=actual,set(wanted)-actual
  j=run('ctest',['ctest','--test-dir','forgecad-release','--output-on-failure','--parallel','1','--output-junit',str(T/'ctest.xml'),'-V','-R','^('+'|'.join(map(re.escape,wanted))+')$']);cases=list(ET.parse(T/'ctest.xml').getroot().iter('testcase'));assert len(cases)==len(wanted);skips=[c.get('name') for c in cases if c.find('skipped') is not None];j.update(selected_tests=wanted,skipped=skips,actual_passed=len(cases)-len(skips));(T/'ctest-result.json').write_text(json.dumps(j,indent=2)+'\n')
