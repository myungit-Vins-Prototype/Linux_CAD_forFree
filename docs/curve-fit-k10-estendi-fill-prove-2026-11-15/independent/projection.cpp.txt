#include "fk_curve_projection.h"
#include "fk_bspline.h"
#include <iostream>
#include <iomanip>
using namespace ForgeCad::Kernel;
int main(){std::cout<<std::setprecision(17);CurvePtr<3> line=std::make_shared<Line<3>>(Vec3(1,2,3),Vec3(2,1,.3));auto trim=std::make_shared<TrimmedCurve<3>>(line,0,10);auto nested=std::make_shared<TrimmedCurve<3>>(trim,2,4);const double m[3][3]={{1.2,.3,.4},{-.2,.8,.2},{.1,0,1.1}};auto affine=Transform3::fromParts(m,Vec3(2,-3,1));auto transformed=std::make_shared<TransformedCurve>(nested,affine);auto source=std::make_shared<TrimmedCurve<3>>(std::make_shared<TransformedCurve>(transformed,Transform3::rotation({},Vec3(1,2,3),.7)),2.1,3.9);Frame3 frame(Vec3(.2,-.3,.4),Vec3(.3,.2,1),Vec3(1,0,0));for(CurvePtr<3> c:{CurvePtr<3>(trim),CurvePtr<3>(nested),CurvePtr<3>(transformed),CurvePtr<3>(source)}){try{auto out=projectCurveAffine(c,{2.2,3.8},frame);double error=0,derivativeError=0;for(int i=0;i<=100;++i){double t=2.2+1.6*i/100.;Vec3 x[2];Vec2 y[2];c->evaluate(t,1,x);out->evaluate(t,1,y);Vec3 q=frame.toLocal(x[0]);error=std::max(error,distance(y[0],Vec2(q.x(),q.y())));derivativeError=std::max(derivativeError,distance(y[1],Vec2(dot(x[1],frame.xDir()),dot(x[1],frame.yDir()))));}std::cout<<"accepted=1 domain="<<out->domain().lo<<","<<out->domain().hi<<" error="<<error<<" derivative_error="<<derivativeError<<"\n";}catch(const std::exception&e){std::cout<<"accepted=0 reason="<<e.what()<<"\n";}}}
