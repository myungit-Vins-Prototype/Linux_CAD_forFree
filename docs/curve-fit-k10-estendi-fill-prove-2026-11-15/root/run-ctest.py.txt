import os,subprocess,time,json
from pathlib import Path
p=Path('/tmp/forgecad-k10-next-20261115/root')
names=(p/'ctest-names.txt').read_text().splitlines()
env=os.environ.copy()
env.update(QT_QPA_PLATFORM='offscreen',QT_QPA_PLATFORMTHEME='generic',QT_STYLE_OVERRIDE='Fusion',QT_QPA_OFFSCREEN_NO_GLX='1',LIBGL_ALWAYS_SOFTWARE='1',XDG_CONFIG_HOME=str(p/'settings'),XDG_DATA_HOME=str(p/'data'),XDG_CACHE_HOME=str(p/'cache'))
for k in ['settings','data','cache']: (p/k).mkdir(exist_ok=True)
t=time.monotonic()
with (p/'ctest.log').open('w') as f:
 r=subprocess.run(['ctest','--test-dir','forgecad-release','--output-on-failure','--parallel','1','-R','^('+'|'.join(names)+')$'],env=env,stdout=f,stderr=subprocess.STDOUT)
(p/'ctest-result.json').write_text(json.dumps(dict(exit=r.returncode,seconds=time.monotonic()-t,tests=names),indent=2))
print('CTest exit',r.returncode,'seconds',round(time.monotonic()-t,2));print((p/'ctest.log').read_text()[-3500:]);raise SystemExit(r.returncode)
