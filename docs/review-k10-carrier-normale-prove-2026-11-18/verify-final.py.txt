from pathlib import Path
import hashlib,json,re,subprocess,os

def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def cmd(args):return subprocess.check_output(args,cwd=root,text=True).strip()
integrated=json.loads((stage/'integrated-hashes.json').read_text())
for f,h in integrated.items():assert sha(root/f)==h,f
assert cmd(['git','rev-parse','HEAD'])=='6294aa37e63c51434fd2754ca7b1398d8a281ffe'

old=json.loads((stage/'previous-seals-verified.json').read_text());count=0
for item in old['seals']:

 for line in p.read_text().splitlines():
  if not line or line.startswith('#'):continue
  h,f=line.split(None,1);assert sha(p.parent/f.lstrip('*'))==h,str(p.parent/f)
  count+=1
assert count==old['payloads']==6099

assert m and json.loads(m.group(1))==p

assert call('cad_surface_mapping.cpp','kernel/fk_surface_mapping.cpp','wrapBodyOnSurface')
assert call('kernel/fk_surface_mapping.cpp','kernel/fk_surface_mapping_bounds.cpp','constantHeightMappingResidualBound')

for f in ['kernel/tests/test_mapping_global_validation.cpp','kernel/tests/test_mapping_normal_validation.cpp']:assert call(f,'kernel/fk_surface_mapping.cpp','wrapBodyOnSurface')
focus=[e for e in edges if (e['source'] in needed or e['target'] in needed) and (e['kind']=='include' or any('constantHeightMappingResidualBound' in s['symbol'] or 'wrapBodyOnSurface' in s['symbol'] for s in e.get('sites',[])))]

commands=json.loads((root/'forgecad-release/compile_commands.json').read_text())
