from verify_tensor_fraction import *
def h(x):return float(x).hex()
def cv(c):return dict(c,type='bspline',knots=list(map(h,c['knots'])),poles=[[h(v) for v in p] for p in c['poles']],weights=list(map(h,c['weights'])))
def sf(s):return dict(s,uPoleCount=s['uCount'],vPoleCount=s['vCount'],uKnots=list(map(h,s['uKnots'])),vKnots=list(map(h,s['vKnots'])),poles=[[h(v) for v in p] for p in s['poles']],weights=list(map(h,s['weights'])))
def evaluate(c,t):
 spans=source_spans(normalize(cv(c)),(F(0),F(1)));s=next(s for s in spans if s['lo']<=t<=s['hi']);f=(t-s['lo'])/(s['hi']-s['lo'])
 def ev(p):
  while len(p)>1:p=[(1-f)*a+f*b for a,b in zip(p,p[1:])]
  return p[0]
 return [ev(p)/ev(s['w']) for p in s['h']]
def surface_point(surface,uv):
 p,q,U,V,nu,nv,P,W=surface_data(sf(surface));u,v=uv
 ku=next(k for k in range(p,nu) if U[k]<U[k+1] and U[k]<=u<=U[k+1]);kv=next(k for k in range(q,nv) if V[k]<V[k+1] and V[k]<=v<=V[k+1]);A=cox(U,p,nu,[u],[F(1)],ku);B=cox(V,q,nv,[v],[F(1)],kv)
 H=[F(0)]*4
 for i in range(nu):
  if not A[i][0]:continue
  for j in range(nv):
   if not B[j][0]:continue
   n=i*nv+j;c=A[i][0]*B[j][0]*W[n]
   for d in range(3):H[d]+=c*P[n][d]
   H[3]+=c
 return [x/H[3] for x in H[:3]]
