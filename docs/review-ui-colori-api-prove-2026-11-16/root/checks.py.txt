from pathlib import Path
import subprocess,json,time,sys,os,re
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/ui-residual-fix/root')
def run(name,cmd,timeout=None,env=None):
 start=time.monotonic()
 with(T/(name+'.log')).open('w') as log:
  try:code=subprocess.run(cmd,cwd=R,stdout=log,stderr=subprocess.STDOUT,timeout=timeout,env=env).returncode
  except subprocess.TimeoutExpired:code=124
 result={'exit':code,'seconds':time.monotonic()-start,'command':cmd};(T/(name+'-result.json')).write_text(json.dumps(result,indent=2)+'\n');print(name,result,flush=True)
 if code:print((T/(name+'.log')).read_text()[-4000:],flush=True);raise SystemExit(code)
 return result
