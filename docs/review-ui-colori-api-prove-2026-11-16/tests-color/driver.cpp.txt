int main(int argc, char **argv) {
    QSurfaceFormat format;
    format.setVersion(3, 3);
    format.setProfile(QSurfaceFormat::CoreProfile);
    format.setDepthBufferSize(24);
    format.setStencilBufferSize(8);
    QSurfaceFormat::setDefaultFormat(format);
    QApplication app(argc, argv);
    QTemporaryDir preferences;
    QSettings::setDefaultFormat(QSettings::IniFormat);
    QSettings::setPath(QSettings::IniFormat, QSettings::UserScope, preferences.path());
    QCoreApplication::setOrganizationName("ForgeCADUiResidualTests");
    QCoreApplication::setApplicationName("isolated");
    qputenv("FORGECAD_RECOVERY_DIR", (preferences.path() + "/recovery").toUtf8());
    try {
        ViewportInteractionTest::uiResidualThemeTests();
        return 0;
    } catch (const std::exception &e) {
        std::cerr << e.what() << std::endl;
        return 1;
    }
}
