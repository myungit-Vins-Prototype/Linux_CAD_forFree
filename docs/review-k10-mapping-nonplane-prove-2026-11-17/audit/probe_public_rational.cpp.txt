void require(bool v,const char*m) {if(!v)throw std::runtime_error(m);}
std::string number(double d) {
    std::ostringstream o;
    o << '"' << std::hexfloat << d << '"';
    return o.str();
}
template <class T> void array(std::ostream &o, const T &v) {
    o << '[';
    bool sep = false;
    for (auto x : v) {
        if (sep)
            o << ',';
        sep = true;
        o << number(x);
    }
    o << ']';
}
template <int N> void vector(std::ostream &o, const Vec<N> &v) {
    o << '[';
    for (int i = 0; i < N; ++i) {
        if (i)
            o << ',';
        o << number(v[i]);
    }
    o << ']';
}
template <int N> void curveJson(std::ostream &o, const CurvePtr<N> &c) {
    if (auto s = std::dynamic_pointer_cast<const BSplineCurve<N>>(c)) {
        o << "{\"type\":\"bspline\",\"degree\":" << s->degree()
          << ",\"knots\":";
        array(o, s->knots());
        o << ",\"poles\":[";
        for (int i = 0; i < s->poleCount(); ++i) {
            if (i)
                o << ',';
            vector(o, s->poles()[i]);
        }
        o << "],\"weights\":[";
        for (int i = 0; i < s->poleCount(); ++i) {
            if (i)
                o << ',';
            o << number(s->weight(i));
        }
        o << "]}";
    } else if (auto l = std::dynamic_pointer_cast<const Line<N>>(c)) {
        o << "{\"type\":\"line\",\"origin\":";
        vector(o, l->origin());
        o << ",\"direction\":";
        vector(o, l->direction());
        o << '}';
    } else if (auto t = std::dynamic_pointer_cast<const TrimmedCurve<N>>(c)) {
        o << "{\"type\":\"trimmed\",\"basis\":";
        curveJson(o, t->basis());
        o << '}';
    } else
        throw std::runtime_error("test export: unsupported curve wrapper");
}
void surfaceJson(std::ostream &o, const BSplineSurface &s) {
    if (s.basis())
        throw std::runtime_error(
            "test export: instance would lose placement rounding");
    o << "{\"uDegree\":" << s.uDegree() << ",\"vDegree\":" << s.vDegree()
      << ",\"uKnots\":";
    array(o, s.uKnots());
    o << ",\"vKnots\":";
    array(o, s.vKnots());
    o << ",\"uPoleCount\":" << s.uPoleCount()
      << ",\"vPoleCount\":" << s.vPoleCount() << ",\"poles\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            if (i || j)
                o << ',';
            vector(o, s.pole(i, j));
        }
    o << "],\"weights\":[";
    for (int i = 0; i < s.uPoleCount(); ++i)
        for (int j = 0; j < s.vPoleCount(); ++j) {
            if (i || j)
                o << ',';
            o << number(s.weight(i, j));
        }
    o << "]}";
}

Body rationalTarget() {
 std::vector<double> knots{0,0,0,1,1,1},weights;std::vector<Vec3> poles;
 const double x[]{0,.25,1},w[]{1,2,3};
 for(int i=0;i<3;++i)for(int j=0;j<3;++j){poles.push_back({x[i],.5*j,0});weights.push_back(w[i]);}
 auto s=std::make_shared<BSplineSurface>(2,2,knots,knots,3,3,poles,weights);
 std::vector<Vec3> points{s->point(0,0),s->point(1,0),s->point(1,1),s->point(0,1)};
 std::vector<Body::BuildEdge> edges{{0,1,s->vIso(0),{0,1},0},{1,2,s->uIso(1),{0,1},0},{3,2,s->vIso(1),{0,1},0},{0,3,s->uIso(0),{0,1},0}};
 auto pc=[](Vec2 a,Vec2 d){return std::make_shared<Line<2>>(a,d);};
 Body::BuildFace f;f.surface=s;f.loops={{{0,true,pc({0,0},{1,0}),0},{1,true,pc({1,0},{0,1}),0},{2,false,pc({0,1},{1,0}),0},{3,false,pc({0,0},{0,1}),0}}};
 return Body::buildSheet(points,edges,{f});
}
int main(int argc,char**argv) {
 try {
  require(argc==2,"output required");
  const Body source=makePlaneSheet(Frame3(),.1),target=rationalTarget();
  const auto before=writeBodyBinary(source),targetBefore=writeBodyBinary(target);
  require(checkBody(source,{true,true}).empty()&&checkBody(target,{true,true}).empty(),"invalid source fixture");
  SurfaceWrapOptions options;options.anchor={.5,.5};SurfaceQuiltMap quilt({{&target,target.faces().front()}},options.anchor);
  double deviation=-1;BodyHistory history;auto output=quilt.bend(source,options,&history,&deviation);
  require(writeBodyBinary(source)==before&&writeBodyBinary(target)==targetBefore,"input mutation");
  require(checkBody(output,{true,true}).empty(),"invalid output");
  std::ofstream out(argv[1]);out<<"{\"scope\":\"public-quilt-rational-final-fins\",\"prefitCap\":"<<number(.8*std::min(options.tolerance,linearResolution()))<<",\"deviation\":"<<number(deviation)<<",\"sourceImmutable\":true,\"targetImmutable\":true,\"records\":[";
  bool first=true;int count=0,certified=0;
  for(auto face:output.faces())for(auto loop:output.face(face).loops)for(auto fid:output.loopFins(loop)) {
   const auto&fin=output.fin(fid);const auto&edge=output.edge(fin.edge);const auto&s=output.face(face).surface;
   const auto r=detail::surfaceCurveResidualBound(edge.curve,s,fin.pcurve,edge.range,.8*std::min(options.tolerance,linearResolution()));
   std::string status=r.status==detail::SurfaceCurveBoundStatus::Certified?"Certified":r.status==detail::SurfaceCurveBoundStatus::Unsupported?"Unsupported":"Failed";
   if(!first)out<<",";first=false;++count;certified+=r.status==detail::SurfaceCurveBoundStatus::Certified;
   out<<"{\"face\":"<<face.index<<",\"fin\":"<<fid.index<<",\"edge\":"<<fin.edge.index<<",\"sense\":"<<fin.sense<<",\"range\":["<<number(edge.range.lo)<<","<<number(edge.range.hi)<<"],\"budget\":"<<number(.8*std::min(options.tolerance,linearResolution()))<<",\"edgeTolerance\":"<<number(edge.tolerance)<<",\"pcurveTolerance\":"<<number(fin.pcurveTolerance)<<",\"runtimeStatus\":\""<<status<<"\",\"runtimeUpper\":"<<number(r.upperBound)<<",\"curve\":";curveJson(out,edge.curve);out<<",\"pcurve\":";curveJson(out,fin.pcurve);out<<",\"surface\":";surfaceJson(out,dynamic_cast<const BSplineSurface&>(*s));out<<"}";
  }
  out<<"]}\n";std::cout<<"public rational fins="<<count<<" Certified="<<certified<<" deviation="<<deviation<<"\n";return 0;
 } catch(const std::exception&e){std::cerr<<e.what()<<"\n";return 1;}
}
