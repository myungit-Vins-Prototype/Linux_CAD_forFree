def trim(p):
 p=list(p)
 while len(p)>1 and p[-1]==0:p.pop()
 return p

def remainder(p,q):
 p=trim(p);q=trim(q)
 while any(p) and len(p)>=len(q):
  shift=len(p)-len(q);f=p[-1]/q[-1]
  for i,x in enumerate(q):p[i+shift]-=f*x
  p=trim(p)
 return p

def gcd(p,q):
 while any(q):p,q=q,remainder(p,q)
 return [x/p[-1] for x in p]

  for k in range(i,n+1):p[k]+=x*math.comb(n,i)*math.comb(n-i,k-i)*(-1)**(k-i)
 return trim(p)
