 checks.append({'pass':bool(ok),'check':msg})
 if not ok:fail.append(msg)

 for s in symbols:check(any(x.endswith('::'+s) for x in nodes[name]['symbols']),'Symbol '+name+': '+s)
for name in ids[11:15]:

 if name.endswith('test_sweep_global.cpp'):
  helpers=['circleRegion','arcSegment','lineSegment','k2Disk']
  check(all(any(x==h or x.endswith('::'+h) for x in nodes[name]['symbols']) for h in helpers),'Global tests resolve local helper definitions')
