def prod(a,b):
 m,n=len(a)-1,len(b)-1
 return [sum((a[i]*b[k-i]*F(comb(m,i)*comb(n,k-i),comb(m+n,k)) for i in range(max(0,k-n),min(m,k)+1)),F(0)) for k in range(m+n+1)]
def derivative(a): return [(len(a)-1)*(a[i+1]-a[i]) for i in range(len(a)-1)]
def at(a,t):
 while len(a)>1:a=[(1-t)*a[i]+t*a[i+1] for i in range(len(a)-1)]
 return a[0]
def sqrtbound(x):
 scale=10**60;n=isqrt(x.numerator*scale*scale//x.denominator)
 return F(n,scale),F(n+1,scale)
def split(a):
 left=[a[0]];right=[a[-1]]
 while len(a)>1:
  a=[(a[i]+a[i+1])/2 for i in range(len(a)-1)];left.append(a[0]);right.append(a[-1])
 return left,right[::-1]
def refine(a,depth):
 if not depth:return [a]
 l,r=split(a);return refine(l,depth-1)+refine(r,depth-1)
def maxabs(a):return max(abs(x) for x in a)
