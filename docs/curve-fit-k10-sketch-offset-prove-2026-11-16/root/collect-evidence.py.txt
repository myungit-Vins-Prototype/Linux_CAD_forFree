from pathlib import Path
import hashlib,shutil,difflib
r=Path('/home/myungit/Linux_CAD_forFree');p=Path('/tmp/forgecad-k10-sketch-offset-20261116');dest=r/'docs/curve-fit-k10-sketch-offset-prove-2026-11-16';dest.mkdir(exist_ok=True)
for folder in ['root','implementation','tests','independent']:
 for f in (p/folder).rglob('*'):
  if not f.is_file() or f.suffix not in {'.py','.cpp','.h','.log','.md','.json','.patch','.csv','.sha256','.txt'} and f.name!='SHA256SUMS':continue
  if any(x in {'settings','data','cache','forgecad-release'} for x in f.relative_to(p/folder).parts):continue
  target=dest/f.relative_to(p)
  if f.suffix in {'.py','.cpp','.h'}:target=target.with_name(target.name+'.txt')
  target.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(f,target)
for name in ['hashes-before.json','architecture-before.json']:shutil.copy2(p/name,dest/name)
diff=[]
for old in sorted((p/'before').rglob('*')):
 if not old.is_file():continue
 rel=old.relative_to(p/'before');new=r/rel
 if new.exists():diff.extend(difflib.unified_diff(old.read_text().splitlines(True),new.read_text().splitlines(True),fromfile='before/'+str(rel),tofile='after/'+str(rel)))
(dest/'change.diff').write_text(''.join(diff))
(dest/'README.md').write_text('''# Prove K10 offset dello schizzo, 16 novembre 2026

corrente nella build Release del repository e usare gli script in `independent/`, che ricompilano

files=sorted(f for f in dest.rglob('*') if f.is_file() and f!=dest/'SHA256SUMS');(dest/'SHA256SUMS').write_text(''.join(hashlib.sha256(f.read_bytes()).hexdigest()+'  '+str(f.relative_to(dest))+'\n' for f in files));print('Collected',len(files),'files')
