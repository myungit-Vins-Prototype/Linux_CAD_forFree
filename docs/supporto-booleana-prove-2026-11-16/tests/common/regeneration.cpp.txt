static QString hash(const QByteArray&b){return QString::fromLatin1(QCryptographicHash::hash(b,QCryptographicHash::Sha256).toHex());}

static QJsonObject shape(const F::ForgeBody&body,const std::string&name){
 QJsonObject j;if(!body){j["geometry"]=false;return j;}j["geometry"]=true;
 const auto&b=*body;const auto issues=K::checkBody(b,{true,true});QJsonArray errors;for(const auto&i:issues)errors.append(QString::fromStdString(K::describe(i.code)+": "+i.message));j["issues"]=errors;
 check(issues.empty(),name+" strict checkBody including loop crossings");
 const auto c=b.counts();j["faces"]=c.faces;j["edges"]=c.edges;j["vertices"]=c.vertices;j["fins"]=c.fins;j["loops"]=c.loops;j["shells"]=c.shells;j["regions"]=c.regions;
 int solidRegions=0;for(auto r:b.regions())solidRegions+=b.region(r).solid; j["solid_regions"]=solidRegions;j["sheet"]=b.isSheet();
 QJsonArray genera;for(auto s:b.shells())genera.append(K::shellGenus(b,s));j["shell_genera"]=genera;
 double vt=0,et=0,pt=0;for(auto v:b.vertices())vt=std::max(vt,b.vertex(v).tolerance);for(auto e:b.edges())et=std::max(et,b.edge(e).tolerance);for(auto f:b.fins())pt=std::max(pt,b.fin(f).pcurveTolerance);j["max_vertex_tolerance"]=vt;j["max_edge_tolerance"]=et;j["max_pcurve_tolerance"]=pt;
 K::Box box;for(auto e:b.edges())box.add(K::curveBox(*b.edge(e).curve,b.edge(e).range));j["bounds_lo"]=point(box.lo);j["bounds_hi"]=point(box.hi);
 QJsonObject types;for(auto face:b.faces()){const QString type=QString::number(int(b.face(face).surface->type()));types[type]=types.value(type).toInt()+1;}j["surface_types"]=types;
 QJsonArray restored;double maxRestoredTolerance=0;
 for(auto id:b.fins()){const auto&fin=b.fin(id);const auto&edge=b.edge(fin.edge);const auto&face=b.face(b.finFace(id));if(edge.curve->type()!=K::CurveType::Ellipse||face.surface->type()!=K::SurfaceType::Cylinder)continue;
  QJsonObject entry{{"fin",id.index},{"edge",fin.edge.index},{"face",b.finFace(id).index},{"first",edge.range.lo},{"last",edge.range.hi},{"pcurve_tolerance",fin.pcurveTolerance}};
  bool ownAffine=fin.pcurve&&fin.pcurve->type()==K::CurveType::BSpline;
  if(ownAffine){const auto&pc=static_cast<const K::BSplineCurve<2>&>(*fin.pcurve);ownAffine=pc.degree()==1&&pc.poleCount()==2&&(pc.weights().empty()||(pc.weights().size()==2&&pc.weights()[0]==1&&pc.weights()[1]==1));entry["degree"]=pc.degree();entry["poles"]=pc.poleCount();entry["pcurve_first"]=pc.domain().lo;entry["pcurve_last"]=pc.domain().hi;ownAffine=ownAffine&&pc.domain().lo==edge.range.lo&&pc.domain().hi==edge.range.hi;}
  entry["affine_in_original_parameter"]=ownAffine;check(ownAffine,name+" restored ellipse/cylinder fin owns affine pcurve in original parameter");check(fin.pcurveTolerance>0&&fin.pcurveTolerance<=K::linearResolution(),name+" restored pcurve carries finite positive certified error within document budget");maxRestoredTolerance=std::max(maxRestoredTolerance,fin.pcurveTolerance);restored.append(entry);
 }
 j["restored_ellipse_cylinder_fins"]=restored;j["max_restored_pcurve_tolerance"]=maxRestoredTolerance;
 const auto m=K::massProperties(b);j["volume"]=m.volume;j["area"]=m.area;j["centroid"]=point(m.centroid);check(std::isfinite(m.volume)&&m.volume>0,name+" finite positive exact-BRep volume");
 j["sha256"]=shapeHash(b);std::cout<<"SHAPE "<<name<<" F="<<c.faces<<" E="<<c.edges<<" V="<<c.vertices<<" shells="<<c.shells<<" solid_regions="<<solidRegions<<" volume="<<m.volume<<" area="<<m.area<<" sha="<<j["sha256"].toString().toStdString()<<std::endl;return j;
}
static QJsonObject genealogy(const F::ForgeBody&result,const std::vector<F::ForgeBody>&sources,const std::string&name){
 const auto h=F::featureHistoryOf(*result);const auto names=F::elementNamesOf(*result);const auto operands=F::genealogyOperandsOf(*result);QJsonObject j;j["operand_count"]=h?h->operandCount():-1;j["registered_operands"]=operands.size();
 check(bool(h)&&bool(names),name+" registered history and persistent names");
 bool valid=bool(h),named=bool(names),facesOrigin=true;int links=0,matchedCarriers=0;QJsonArray sourceIDs;
 for(const auto&o:operands){sourceIDs.append(double(o.featureId));valid=valid&&bool(o.body.lock());}
 j["operand_feature_ids"]=sourceIDs;
 for(int kind=0;kind<3;++kind){std::vector<int> ids;if(kind==0)for(auto v:result->vertices())ids.push_back(v.index);if(kind==1)for(auto e:result->edges())ids.push_back(e.index);if(kind==2)for(auto f:result->faces())ids.push_back(f.index);
  for(int index:ids){if(names)named=named&&!names->primary(K::EntityKind(kind),index).empty();if(!h)continue;const auto&origins=h->origins(K::EntityKind(kind),index);if(kind==2)facesOrigin=facesOrigin&&!origins.empty();for(const auto&o:origins){++links;valid=valid&&o.operand>=0&&o.operand<operands.size();if(o.operand<0||o.operand>=operands.size())continue;auto b=operands[o.operand].body.lock();valid=valid&&b&&o.index>=0&&o.index<h->operandSize(o.operand,o.kind);if(!b)continue;if(o.kind==K::EntityKind::Face)valid=valid&&b->contains(K::FaceId(o.index));if(o.kind==K::EntityKind::Edge)valid=valid&&b->contains(K::EdgeId(o.index));if(o.kind==K::EntityKind::Vertex)valid=valid&&b->contains(K::VertexId(o.index));}}
 }
 for(auto f:result->faces()){bool found=false;for(const auto&input:sources)for(auto old:input->faces())found=found||K::sameSurfaceParameterization(*result->face(f).surface,*input->face(old).surface)||K::sameSurface(*result->face(f).surface,*input->face(old).surface,K::linearResolution());matchedCarriers+=found;}
 check(valid,name+" history links refer to live operand entities");check(named,name+" all live face/edge/vertex names nonempty");check(facesOrigin,name+" every resulting face has operand provenance");check(matchedCarriers==int(result->faces().size()),name+" every face preserves an original surface carrier");j["valid_links"]=valid;j["all_named"]=named;j["faces_with_origins"]=facesOrigin;j["links"]=links;j["original_surface_carriers"]=matchedCarriers;return j;
}
static std::vector<K::Vec3> witnesses(const K::Body&b){std::vector<K::Vec3> p;for(auto v:b.vertices())p.push_back(b.vertex(v).point);for(auto e:b.edges())for(double f:{.25,.5,.75})p.push_back(b.edge(e).curve->point(b.edge(e).range.lo+f*b.edge(e).range.length()));for(auto f:b.faces()){K::Vec3 q;if(K::faceInteriorPoint(b,f,q))p.push_back(q);}return p;}
static QJsonObject geometricWitnesses(const F::ForgeBody&out,const std::vector<F::ForgeBody>&sources,const std::string&name){
 K::SolidClassifier target(*out,K::linearResolution());int inputCount=0,lost=0,extra=0,outputCount=0;
 for(const auto&s:sources)for(const auto&p:witnesses(*s)){++inputCount;if(target.classify(p)==K::PointLocation::Outside)++lost;}
 std::vector<std::unique_ptr<K::SolidClassifier>> classifiers;for(const auto&s:sources)classifiers.push_back(std::make_unique<K::SolidClassifier>(*s,K::linearResolution()));
 for(const auto&p:witnesses(*out)){++outputCount;bool found=false;for(const auto&c:classifiers)found=found||c->classify(p)!=K::PointLocation::Outside;if(!found)++extra;}
 check(lost==0,name+" source boundary witnesses retained by union");check(extra==0,name+" output boundary witnesses belong to input union");std::cout<<"WITNESS "<<name<<" inputs="<<inputCount<<" lost="<<lost<<" outputs="<<outputCount<<" extra="<<extra<<std::endl;return {{"input_count",inputCount},{"lost",lost},{"output_count",outputCount},{"extra",extra},{"continuous_certificate",false}};
}
static QJsonObject operation(const DocumentState&doc,int a,int b,const QVector<int>&tools,const std::string&label){
 const auto before=[&]{QStringList s;for(const auto&x:doc.extrusions)s<< (x.forgeBody?shapeHash(*x.forgeBody):QString());return s;}();

  double total=0,largest=0;for(auto x:inputs){const double v=K::massProperties(*x).volume;total+=v;largest=std::max(largest,v);}const double volume=j["shape"].toObject()["volume"].toDouble();const double quadratureBudget=1e-10*std::max(1.,total);check(volume>=largest-quadratureBudget&&volume<=total+quadratureBudget,label+" union volume is between maximum operand and sum");j["volume_sum_inputs"]=total;j["volume_overlap_implied"]=total-volume;

 }
 return j;
}

static QJsonObject viewportOperation(DocumentState doc,int a,int b,const QVector<int>&tools,const std::string&label) {
 // La rigenerazione senza cache e' gia' verificata separatamente. Qui carica
 // gli stessi B-rep immutabili per isolare il vero entrypoint applicativo.
 for(auto&f:doc.extrusions)if(f.forgeBody)f.cachedGeometry=true;
 CadViewport view;view.loadDocument(doc);const auto before=view.recoveryDocument();const auto beforeHash=definitionHash(before);
 QVector<int> allTools{b};allTools+=tools;
 const auto error=view.createBoolean(BooleanOperation::Union,a,allTools,QString::fromStdString(label));
 auto after=view.recoveryDocument();const bool success=error.isEmpty();
 QJsonObject j{{"label",QString::fromStdString(label)},{"first",a},{"second",b},{"success",success},{"error",error}};
 QJsonArray t;for(int x:tools)t.append(x);j["tools"]=t;
 std::cout<<"APP_OPERATION "<<label<<" success="<<success<<" features_before="<<before.extrusions.size()<<" features_after="<<after.extrusions.size()<<" error="<<error.toStdString()<<std::endl;
 bool immutable=true;for(int i=0;i<before.extrusions.size();++i)if(before.extrusions[i].forgeBody)immutable=immutable&&after.extrusions[i].forgeBody&&shapeHash(*before.extrusions[i].forgeBody)==shapeHash(*after.extrusions[i].forgeBody);
 check(immutable,label+" app preserves immutable operand geometry");
 if(!success){check(after.extrusions.size()==before.extrusions.size()&&definitionHash(after)==beforeHash,label+" failed app operation is atomic for definition, visibility and history");return j;}
 check(after.extrusions.size()==before.extrusions.size()+1,label+" app appends exactly one feature");
 const auto&f=after.extrusions.back();check(f.operation==int(BooleanOperation::Union)&&f.firstBody==a&&f.secondBody==b&&f.booleanTools==tools&&f.featureId==7,label+" app result retains intended operands and new feature identity");
 check(!after.extrusions[a].visible&&!after.extrusions[b].visible,label+" app hides the consumed operands");
 std::vector<F::ForgeBody> inputs{after.extrusions[a].forgeBody,after.extrusions[b].forgeBody};for(int x:tools)inputs.push_back(after.extrusions[x].forgeBody);
 j["shape"]=shape(f.forgeBody,label);j["genealogy"]=genealogy(f.forgeBody,inputs,label);j["witnesses"]=geometricWitnesses(f.forgeBody,inputs,label);
 const auto resultingHash=shapeHash(*f.forgeBody);view.undo();check(view.extrusions().size()==before.extrusions.size(),label+" app Undo restores feature count");view.redo();check(view.extrusions().size()==before.extrusions.size()+1&&shapeHash(*view.extrusions().back().forgeBody)==resultingHash,label+" app Redo restores identical union geometry");return j;
}
