#!/usr/bin/env python3
"""Independent exact coefficient audit; no kernel calls or sampled geometry.
Inputs are round-trip decimal getters from geometry.log generated by core Sol.
Every decimal is converted to its binary64 value and then exact Fraction.
"""
from fractions import Fraction as F
from math import isqrt
from pathlib import Path
import hashlib,json

def binary(s): return F.from_float(float(s))
def vec(ss): return tuple(binary(s) for s in ss)
def dot(a,b): return sum((x*y for x,y in zip(a,b)),F())
def sub(a,b): return tuple(x-y for x,y in zip(a,b))
def sqrt_outward(q, digits=80):
    assert q >= 0
    scale=10**digits
    lo=isqrt(q.numerator*scale*scale//q.denominator)
    # Exact integer comparisons, including a possible exact square.
    assert F(lo,scale)**2 <= q
    hi=lo if F(lo,scale)**2==q else lo+1
    assert F(hi,scale)**2 >= q
    return F(lo,scale),F(hi,scale)
def dec(q,digits=60):
    # Truncation for display only; rational exact values are also recorded.
    sign='-' if q<0 else '';q=abs(q);z=q.numerator*10**digits//q.denominator
    s=str(z).rjust(digits+1,'0');return sign+s[:-digits]+'.'+s[-digits:]
def bounds(q): return {'exact_fraction':str(q),'decimal_truncated':dec(q)}

center=vec(['-130.37600593762954','75','135.09900801728884'])
xaxis=vec(['-0.25463541661500666','0','-0.96703712679777809'])
yaxis=vec(['0','-1','0'])
rx=binary('23.996850146381604');ry=binary('10')
origin=vec(['-130','42.5','135']);axis=vec(['0','1','0']);radius=binary('24.000000000000007')
assert dot(axis,axis)==1
# Project perpendicular to the exact unit cylinder axis.
def perpendicular(v):return sub(v,tuple(dot(v,axis)*d for d in axis))
c=perpendicular(sub(center,origin));a=perpendicular(tuple(rx*d for d in xaxis));b=perpendicular(tuple(ry*d for d in yaxis))
assert dot(b,b)==0
A=dot(a,a); B=dot(c,a); C=dot(c,c)-radius*radius
assert A>0
# rho(u)^2 - R^2 = A*cos(u)^2 + 2*B*cos(u) + C.
# Convex on [-1,1], hence maximum equals max(q(-1),q(1)).
qminus=A-2*B+C;qplus=A+2*B+C
peak=max(qminus,qplus)
assert qminus>0 and qplus>0
radlow,radhigh=sqrt_outward(radius*radius+peak)
xmin=-B/A
assert -1<xmin<1
qmin=C-B*B/A
minlow,minhigh=sqrt_outward(radius*radius+qmin)
# Geometric cap protrusion is strictly positive, globally below 2e-11 mm.
assert 0<radlow-radius<=radhigh-radius<F(2,10**11)
endpoints={}
for name,q in [('cos_minus_one',qminus),('cos_plus_one',qplus)]:
 lo,hi=sqrt_outward(radius*radius+q)
 endpoints[name]={'squared_radial_excess':bounds(q),'radial_excess_lower':bounds(lo-radius),'radial_excess_upper':bounds(hi-radius)}
report={'method':'Exact Fraction polynomial in cos(u); convexity; integer-square-root outward bounds',
'input_source':'geometry.log: getters of actual stored analytic cap ellipse and cylinder',
'input_source_sha256':hashlib.sha256(Path('/tmp/supporto-review/geometry.log').read_bytes()).hexdigest(),
'input_binary64_hex':{'center':[float(x).hex() for x in center],'xaxis':[float(x).hex() for x in xaxis],'yaxis':[float(x).hex() for x in yaxis],'xradius':float(rx).hex(),'yradius':float(ry).hex(),'cylinder_origin':[float(x).hex() for x in origin],'cylinder_axis':[float(x).hex() for x in axis],'cylinder_radius':float(radius).hex()},
'quadratic':{'A':bounds(A),'B':bounds(B),'C':bounds(C)},'endpoints':endpoints,
'global_radial_excess_upper':bounds(radhigh-radius),'global_radial_excess_lower_at_peak':bounds(radlow-radius),
'global_radial_minimum_interval':[bounds(minlow-radius),bounds(minhigh-radius)],
'conclusions':['Both analytic cap extremities are strictly outside the cylinder for these exact binary64 coefficients.','Maximum continuous radial protrusion of entire analytic cap ellipse is below 2e-11 mm.','This is more than 5000 times below document linear resolution 1e-7 mm.','This certificate concerns the input analytic ellipse/cylinder, not arbitrary loft interior surfaces or Boolean output.']}
Path('/tmp/supporto-review/cap-radial-fraction.json').write_text(json.dumps(report,indent=2)+'\n')
for k,v in endpoints.items():print(k,'radial excess in [',dec(F(v['radial_excess_lower']['exact_fraction'])),',',dec(F(v['radial_excess_upper']['exact_fraction'])),'] mm')
print('CONTINUOUS MAX UPPER',dec(radhigh-radius),'mm')
print('PASS exact signs, convex global maximum, outward sqrt, bound <2e-11 mm')

# Continuous certificate for the TWO ACTUAL tiny source-edge ranges retained
# by the trace, using the same original ellipse coefficients. Pi enclosure
# uses Machin's identity with alternating rational arctan sums.
def atan_outward(x,terms=40):
 s=sum(((-1)**k*x**(2*k+1)/F(2*k+1) for k in range(terms)),F())
 nxt=(-1)**terms*x**(2*terms+1)/F(2*terms+1)
 return min(s,s+nxt),max(s,s+nxt)
a0,a1=atan_outward(F(1,5));b0,b1=atan_outward(F(1,239))
pi_lo=16*a0-4*b1;pi_hi=16*a1-4*b0
edge2_lo=binary('6.2831840222190092');edge2_hi=binary('6.2831853071795862')
edge5_lo=F();edge5_hi=binary('1.2841783501532532e-06')
assert edge2_lo<edge2_hi<2*pi_lo
# Every angle in the edge2 interval has cos(t)=cos(2*pi-t).
delta=max(2*pi_hi-edge2_lo,edge5_hi)
assert 0<delta<F(1,100)
x_lower=1-delta*delta/2 # cos(x)>=1-x^2/2, no floating point evaluation
assert A*x_lower+B>0 # q is strictly increasing on [x_lower,1]
q_chain_min=A*x_lower*x_lower+2*B*x_lower+C
rho_lo,_=sqrt_outward(radius*radius+q_chain_min)
chain_lower=rho_lo-radius;chain_upper=radhigh-radius
assert -F(2,10**11)<chain_lower<chain_upper<F(2,10**11)
chain_report={'source_ranges_binary64_hex':{'edge2':[float(edge2_lo).hex(),float(edge2_hi).hex()],'edge5':[float(edge5_lo).hex(),float(edge5_hi).hex()]},
'pi_method':'Machin identity; 40-term alternating rational arctan bounds',
'pi_interval':[str(pi_lo),str(pi_hi)],'max_reduced_angle_upper':bounds(delta),'cos_lower':bounds(x_lower),
'continuous_radial_deviation_lower':bounds(chain_lower),'continuous_radial_deviation_upper':bounds(chain_upper),
'conclusion':'For every parameter of the two retained original source-edge spans, absolute distance to the infinite cylinder carrier is below 2e-11 mm. No sampled endpoints or production evaluator are used.'}
Path('/tmp/supporto-review/cap-chain-fraction.json').write_text(json.dumps(chain_report,indent=2)+'\n')
print('RETAINED CHAIN CONTINUOUS RADIAL INTERVAL [',dec(chain_lower),',',dec(chain_upper),'] mm')
print('PASS source-edge chain is continuously within 2e-11 mm of cylinder carrier')
