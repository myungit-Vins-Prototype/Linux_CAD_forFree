+                             bool selecting, QString &requestedKey) {
+    const QString key = ForgeCad::Gui::FeaturePreviewBuilder::previewKey(definition, scope.index);
+    if (!selecting) {
+        requestedKey = key;

+        requestedKey.clear();
+    }

+    return invalidated;
