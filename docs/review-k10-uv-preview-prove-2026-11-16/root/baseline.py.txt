from pathlib import Path
import json,hashlib,subprocess,shutil,datetime
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/k10-uv-review/root')
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
names=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','-z'],cwd=R).decode().split('\0')
hashes={f:sha(R/f) for f in sorted(set(names)) if f and (R/f).is_file() and not (R/f).is_symlink()}
(T/'hashes-before.json').write_text(json.dumps(hashes,indent=2)+'\n')
focus=['CLAUDE.md','docs/code-review-2026-11-15.md','kernel/fk_offset.cpp','kernel/fk_curve_fit_bounds.h','kernel/fk_curve_fit_bounds.cpp','kernel/CMakeLists.txt','tools/generate_architecture.py','forgeCad2026_dialogs_mapping.cpp','forgeCad2026_dialogs_surfaces.cpp']
for f in focus:
 d=T/'before'/(f+'.txt');d.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(R/f,d)
g=json.loads((R/'docs/architecture/graph.json').read_text());nodes={n['id']:n for n in g['nodes']}
related=set(focus)|{'kernel/fk_affine_surface_fit.cpp','kernel/fk_topology.cpp','kernel/fk_curve_projection.cpp','forgeCad2026_feature_preview.cpp','forgeCad2026_dialogs_common.cpp'}
edges=[e for e in g['edges'] if e['source'] in related or e['target'] in related]
(T/'architecture-before.json').write_text(json.dumps(dict(nodes={f:nodes[f] for f in related if f in nodes},edges=edges),indent=2)+'\n')
print('Baseline',len(hashes),'file;',len(edges),'archi consultati')
for f in ['kernel/fk_offset.cpp','kernel/fk_curve_fit_bounds.cpp','kernel/fk_curve_projection.cpp']:
 print(f,nodes[f]['description']);print('In/out', {k:(sum(e['target']==f and e['kind']==k for e in edges),sum(e['source']==f and e['kind']==k for e in edges)) for k in ['call','include','reference']})
for f in ['kernel/fk_offset.cpp','kernel/fk_curve_fit_bounds.cpp']:
 print('Resolved dependencies',f,sorted(set((e['target'],s['symbol']) for e in edges if e['source']==f and e['kind']=='call' for s in e['sites'])))
