from pathlib import Path
import subprocess,os,signal,tempfile,time,json
R=Path('/home/myungit/Linux_CAD_forFree');T=Path('/tmp/boolean-order-review/root')
with tempfile.TemporaryDirectory(prefix='forgecad-theme-parent-debug-') as d:
 env=os.environ.copy();env.update(QT_QPA_PLATFORM='wayland',QT_QPA_PLATFORMTHEME='',QT_STYLE_OVERRIDE='Fusion',XDG_CONFIG_HOME=d,FORGECAD_RECOVERY_DIR=d+'/recovery')

 print('debug exit',code, 'interrupted',interrupted)
