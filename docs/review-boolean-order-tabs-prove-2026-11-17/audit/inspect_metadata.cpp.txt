#include "fk_body_io.h"
#include <fstream>
#include <iostream>
#include <iomanip>
#include <iterator>
#include <cmath>
#include <map>
using namespace ForgeCad::Kernel;
int main(int argc,char**argv) {
 try {
  std::cout<<"[";
  for(int arg=1;arg<argc;++arg) {
   std::ifstream f(argv[arg],std::ios::binary);
   if(!f)throw std::runtime_error("missing staged body");
   Body b=readBodyBinary(std::string(std::istreambuf_iterator<char>(f),{}));
   if(arg>1)std::cout<<",";
   std::cout<<"{\"inputIndex\":"<<arg-1<<",\"faces\":"<<b.faces().size()<<",\"vertices\":[";
   bool first=true;
   const auto value=[](double x){if(!std::isfinite(x)||x<0)throw std::runtime_error("invalid tolerance");std::cout<<'"'<<std::hexfloat<<x<<std::defaultfloat<<'"';};
   for(auto v:b.vertices()) {if(!first)std::cout<<",";first=false;std::cout<<"{\"id\":"<<v.index<<",\"tol\":";value(b.vertex(v).tolerance);std::cout<<"}";}
   std::cout<<"],\"edges\":[";first=true;
   for(auto e:b.edges()) {if(!first)std::cout<<",";first=false;const auto&ed=b.edge(e);std::cout<<"{\"id\":"<<e.index<<",\"tol\":";value(ed.tolerance);std::cout<<",\"curveType\":"<<int(ed.curve->type())<<"}";}
   std::cout<<"],\"fins\":[";first=true;
   for(auto f:b.fins()) {if(!first)std::cout<<",";first=false;const auto&fn=b.fin(f);std::cout<<"{\"id\":"<<f.index<<",\"edge\":"<<fn.edge.index<<",\"face\":"<<b.finFace(f).index<<",\"tol\":";value(fn.pcurveTolerance);std::cout<<",\"sense\":"<<fn.sense<<",\"pcurveType\":"<<(fn.pcurve?int(fn.pcurve->type()):-1)<<"}";}
   std::cout<<"]}";
  }
  std::cout<<"]\n";
 }catch(const std::exception&e) {std::cerr<<e.what()<<"\n";return 1;}
}
