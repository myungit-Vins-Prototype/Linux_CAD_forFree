+bool knotsValid(const std::vector<double> &k, int p, int n) {
+    if (n <= p || k.size() != std::size_t(n + p + 1)) return false;
+    for (std::size_t i = 0; i < k.size(); ++i) {
+        cancellationPoint();
+        if (!std::isfinite(k[i]) || (i && k[i] < k[i - 1])) return false;
+    }
+    if (!(k[std::size_t(p)] < k[std::size_t(n)])) return false;
+    for (int i = p + 1, repeat = 1; i < n; ++i) {
+        repeat = k[std::size_t(i)] == k[std::size_t(i - 1)] ? repeat + 1 : 1;
+        if (repeat > p) return false;
+    }
+    return true;
+}
+void validate(const BSplineSurface &s, bool emitted) {
+    if (s.basis()) throw Unsupported{}; // Do not read recursively or round transformed poles.
+    if ((emitted && (s.uDegree() != 3 || s.vDegree() != 3)) ||
+        (!emitted && (s.uDegree() < 2 || s.uDegree() > 3 || s.vDegree() < 2 || s.vDegree() > 3)) ||
+        s.isUPeriodic() || s.isVPeriodic()) throw Unsupported{};
+    const auto nu = s.uPoleCount(), nv = s.vPoleCount();
+    if (nu <= 0 || nv <= 0 || std::size_t(nu) > maxPoles / std::size_t(nv)) throw ProofFailed{};
+    if (!knotsValid(s.uKnots(), s.uDegree(), nu) || !knotsValid(s.vKnots(), s.vDegree(), nv)) throw ProofFailed{};
+    const double w = s.weight(0, 0);
+    if (!std::isfinite(w) || !(w > 0)) throw ProofFailed{};
+    bool constant = true;
+    for (int i = 0; i < nu; ++i) for (int j = 0; j < nv; ++j) {
+        cancellationPoint();
+        const double wi = s.weight(i, j);
+        if (!std::isfinite(wi) || !(wi > 0)) throw ProofFailed{};
+        constant = constant && wi == w;
+        point(s.pole(i, j));
+    }
+    if (!constant) throw Unsupported{};
+}
+std::vector<int> cells(const std::vector<double> &k, int p, int n, I query) {
+    if (!finite(query) || query.lo < k[std::size_t(p)] || query.hi > k[std::size_t(n)]) throw ProofFailed{};
+    std::vector<int> out;
+    // Begin on the left of the FIRST equal knot, preserving closed-cell
+    // coverage at repeated knots; otherwise begin at the containing cell.
+    const int first = std::max(p, int(std::lower_bound(k.begin() + p, k.begin() + n + 1, query.lo) - k.begin()) - 1);
+    for (int i = first; i < n && k[std::size_t(i)] <= query.hi; ++i) {
+        cancellationPoint();
+        if (k[std::size_t(i)] < k[std::size_t(i + 1)] && query.lo <= k[std::size_t(i + 1)]) {
+            if (out.size() == maxCells) throw ProofFailed{};
+            out.push_back(i);
+        }
+    }
+    if (out.empty()) throw ProofFailed{};
+    return out;
+}
+// Each active target cell polynomial is enclosed on the WHOLE source rectangle.
+// This includes extrapolation outside that cell. Taking the maximum over ALL
+// intersected closed cells is conservative; selecting a midpoint cell is not.
+VectorTensor tensor(const BSplineSurface &s, int cu, int cv, const P &u, const P &v,
+                    const IP &origin, bool swapped) {
+    auto bu = basis(s.uKnots(), cu, u, s.uDegree());
+    auto bv = basis(s.vKnots(), cv, v, s.vDegree());
+    for (auto &b : bu) b = elevated(std::move(b), 3);
+    for (auto &b : bv) b = elevated(std::move(b), 3);
+    VectorTensor out{};
+    for (int i = 0; i <= s.uDegree(); ++i) for (int j = 0; j <= s.vDegree(); ++j) {
+        cancellationPoint();
+        const auto p = point(s.pole(cu - s.uDegree() + i, cv - s.vDegree() + j));
+        for (int a = 0; a < 4; ++a) for (int b = 0; b < 4; ++b) {
+            const I weight = bu[std::size_t(i)][std::size_t(a)] * bv[std::size_t(j)][std::size_t(b)];
+            const auto index = std::size_t(swapped ? b * 4 + a : a * 4 + b);
+            for (int axis = 0; axis < 3; ++axis)
+                out[std::size_t(axis)][index] = out[std::size_t(axis)][index] + (p[std::size_t(axis)] - origin[std::size_t(axis)]) * weight;
+        }
+    }
+    return out;
+}
+struct Rectangle { Interval u, v; };
+I enclosure(const P &p) { return hull(p); }
+double node(const BSplineSurface &target, const BSplineSurface &image,
+            const std::array<Affine, 2> &map, int iu, int iv, Rectangle r,
+            double budget, int depth, Work &work, const IP &origin) {
+    work.visit();
+    const bool swapped = exact(map[0].du, 0);
+    const P uq{at(map[0], r.u.lo, r.v.lo), at(map[0], r.u.hi, r.v.hi)};
+    const P vq{at(map[1], r.u.lo, r.v.lo), at(map[1], r.u.hi, r.v.hi)};
+    const auto uc = cells(target.uKnots(), target.uDegree(), target.uPoleCount(), enclosure(uq));
+    const auto vc = cells(target.vKnots(), target.vDegree(), target.vPoleCount(), enclosure(vq));
+    if (uc.size() > maxCells / vc.size()) throw ProofFailed{};
+    const auto actual = tensor(image, iu, iv, P{I(r.u.lo), I(r.u.hi)}, P{I(r.v.lo), I(r.v.hi)}, origin, false);
+    double upper = 0;
+    for (int tu : uc) for (int tv : vc) {
+        work.visit();
+        const auto wanted = tensor(target, tu, tv, uq, vq, origin, swapped);
+        I squared;
+        for (int a = 0; a < 3; ++a) {
+            double bound = 0;
+            for (std::size_t k = 0; k < 16; ++k) {
+                const I d = actual[std::size_t(a)][k] - wanted[std::size_t(a)][k];
+                if (!finite(d)) throw ProofFailed{};
+                bound = std::max(bound, std::max(std::abs(d.lo), std::abs(d.hi)));
+            }
+            squared = squared + I(bound) * I(bound);
+        }
+        if (!finite(squared)) throw ProofFailed{};
+        upper = std::max(upper, std::nextafter(std::sqrt(squared.hi), infinity));
+    }
+    if (upper <= budget) return upper;
+    if (depth == maxDepth) throw ProofFailed{};
+    // Alternate source axes, retaining the original anchors in every Cox call.
+    const bool splitU = depth % 2 == 0;
+    const Interval q = splitU ? r.u : r.v;
+    const double mid = .5 * q.lo + .5 * q.hi;
+    if (!(q.lo < mid && mid < q.hi)) throw ProofFailed{};
+    Rectangle left = r, right = r;
+    (splitU ? left.u : left.v).hi = mid;
+    (splitU ? right.u : right.v).lo = mid;
+    return std::max(node(target, image, map, iu, iv, left, budget, depth + 1, work, origin),
+                    node(target, image, map, iu, iv, right, budget, depth + 1, work, origin));
+}
+std::vector<double> cuts(const std::vector<double> &knots, Interval range) {
+    std::vector<double> out{range.lo};
+    for (double k : knots) if (range.lo < k && k < range.hi && out.back() != k) out.push_back(k);
+    out.push_back(range.hi);
+    return out;
+}
+int containing(const std::vector<double> &k, int degree, int count, Interval r) {
+    int cell = int(std::upper_bound(k.begin(), k.end(), r.lo) - k.begin()) - 1;
+    cell = std::clamp(cell, degree, count - 1);
+    if (!(k[std::size_t(cell)] <= r.lo && r.hi <= k[std::size_t(cell + 1)])) throw ProofFailed{};
+    return cell;
+}
+}
+SurfaceMappingBoundResult zeroHeightMappingResidualBound(
+    const Surface &source, const Surface &target, const SurfaceWrapOptions &options,
+    double scaleU, double scaleV, const Surface &image, Interval u, Interval v, double budget) {
+    cancellationPoint();
+    try {
+        if (!std::isfinite(u.lo) || !std::isfinite(u.hi) || !(u.lo < u.hi) ||
+            !std::isfinite(v.lo) || !std::isfinite(v.hi) || !(v.lo < v.hi) ||
+            !std::isfinite(budget) || !(budget > 0) || !std::isfinite(scaleU) || !(scaleU > 0) ||
+            !std::isfinite(scaleV) || !(scaleV > 0) || !std::isfinite(options.angle) ||
+            !std::isfinite(options.normalOffset) || !std::isfinite(options.anchor.x()) ||
+            !std::isfinite(options.anchor.y()) || !std::isfinite(options.shift.x()) || !std::isfinite(options.shift.y())) throw ProofFailed{};
+        const auto *plane = dynamic_cast<const Plane *>(&source);
+        const auto *t = dynamic_cast<const BSplineSurface *>(&target);
+        const auto *f = dynamic_cast<const BSplineSurface *>(&image);
+        if (!plane || !t || !f || options.rigid) throw Unsupported{};
+        const auto map = mapping(*plane, options, scaleU, scaleV);
+        validate(*t, false); validate(*f, true);
+        const auto &uk = f->uKnots(), &vk = f->vKnots();
+        if (u.lo < uk[3] || u.hi > uk[std::size_t(f->uPoleCount())] ||
+            v.lo < vk[3] || v.hi > vk[std::size_t(f->vPoleCount())]) throw ProofFailed{};
+        const auto us = cuts(uk, u), vs = cuts(vk, v);
+        if ((us.size() - 1) > maxVisits / (vs.size() - 1)) throw ProofFailed{};
+        const auto origin = point(t->pole(0, 0)); // Center only; geometry remains original intervals.
+        Work work; double upper = 0;
+        for (std::size_t i = 1; i < us.size(); ++i) for (std::size_t j = 1; j < vs.size(); ++j) {
+            Rectangle r{{us[i - 1], us[i]}, {vs[j - 1], vs[j]}};
+            upper = std::max(upper, node(*t, *f, map,
+                containing(uk, 3, f->uPoleCount(), r.u), containing(vk, 3, f->vPoleCount(), r.v),
+                r, budget, 0, work, origin));
+        }
+        return {SurfaceMappingBoundStatus::Certified, upper};
+    } catch (const Unsupported &) { return {}; }
+      catch (const ProofFailed &) { return {SurfaceMappingBoundStatus::Failed, infinity}; }
+}

+FK_TEST(MappingBoundsInvalidInputsFailAndNonzeroOffsetIsUnsupported) {
+    const Plane source{Frame3()};
+    const auto target = flatSurface(2), image = flatSurface(3);
+    const double infinity = std::numeric_limits<double>::infinity();
+    const double nan = std::numeric_limits<double>::quiet_NaN();
+    for (double cap : {0., -1., infinity, nan})
+        FK_CHECK(bound(target, image, source, {}, cap).status == Status::Failed);
+    for (Interval query : {Interval{0, infinity}, Interval{nan, 1}, Interval{1, 0}, Interval{-.01, 1}})
+        FK_CHECK(bound(target, image, source, {}, budget, query).status == Status::Failed);
+    SurfaceWrapOptions options;
+    options.shift = {nan, 0};
+    FK_CHECK(bound(target, image, source, options).status == Status::Failed);
+    options = {};
+    options.normalOffset = .001;
+    FK_CHECK(bound(target, image, source, options).status == Status::Unsupported);
+    // Variable rational weights remain outside this polynomial-only delta.
+    std::vector<Vec3> poles;
+    for (int i = 0; i < 3; ++i) for (int j = 0; j < 3; ++j)
+        poles.emplace_back(i / 2., j / 2., 0);
+    std::vector<double> weights(9, 1.);
+    weights[4] = 2.;
+    const BSplineSurface rational(2, 2, clamped(2), clamped(2), 3, 3, poles, weights);
+    FK_CHECK(bound(rational, image, source).status == Status::Unsupported);
+}
