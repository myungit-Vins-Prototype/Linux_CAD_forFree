#!/usr/bin/env python3
"""Exact point/global-control-hull separation for reconstructed old module."""
from certify import *

def sqrt_down(f):
    v=sqrt(float(f))
    while F.from_float(v)**2>f:v=nextafter(v,-inf)
    return v

def evaluate(coeff,t):
    while len(coeff)>1:coeff=[(1-t)*a+t*b for a,b in zip(coeff,coeff[1:])]
    return coeff[0]

def main():
    results=[]
    for r in map(json.loads,open(sys.argv[1])):
        if 'name' not in r:continue
        src=source_data(r['source'],vector(r['origin']),vector(r['x']),vector(r['y']),vector(r['d']))
        A,W,nd,props=src;axis=vector(r['y']);hull=[dot(axis,vector(p)) for piece in r['pieces'] for p in piece['curve']['poles']]
        low,high=min(hull),max(hull);witnesses=[]
        for t in [F(8793,10000),F(1207,10000)]:
            projected=[evaluate(a,t)/(nd*evaluate(W,t)) for a in A]
            value=dot(axis,projected);gap=max(low-value,value-high,F(0));sq=gap**2/dot(axis,axis)
            witnesses.append((sq,t,projected,gap))
        sq,t,p,gap=max(witnesses,key=lambda v:v[0]);assert sq>F.from_float(1e-7)**2
        result={'name':r['name'],'witness_parameter':str(t),'distance_lower':sqrt_down(sq),'separation_squared_numerator':str(sq.numerator),'separation_squared_denominator':str(sq.denominator),'source_properties':props,'baseline_piece_spans':sum((len(c['curve']['poles'])-1)//3 for c in r['pieces']),'published_deviation':float.fromhex(r['pieces'][0]['deviation'])}
        results.append(result);print(r['name'],result['distance_lower'],flush=True)
    out={'module_baseline_scope':'Git fk_project.cpp/h snapshot relinked before current Release kernel archive; not the historical whole-kernel archive','geometry_witnesses':len(results),'minimum_geometric_distance_lower':min(r['distance_lower'] for r in results),'results':results}
    json.dump(out,open(sys.argv[2],'w'),indent=2);print('SUMMARY',json.dumps({k:v for k,v in out.items() if k!='results'}))
if __name__=='__main__':main()
