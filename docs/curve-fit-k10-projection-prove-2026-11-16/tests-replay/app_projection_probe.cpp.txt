SketchObject source(bool nurbs,double gap){
 SketchObject s;s.customFrame=true;s.frame.origin[2]=1.;
 if(!nurbs){s.segments.append({QPointF(0,0),QPointF(1,0)});s.segments.append({QPointF(1+gap,0),QPointF(2+gap,0)});}
 else for(int piece=0;piece<2;++piece){CurveObject c;c.tool=DrawingTool::Converted;c.degree=27;for(int i=0;i<28;++i)c.controlPoints.append(QPointF(piece*(1.+gap)+i/27.,z27[i]));for(int i=0;i<28;++i)c.knots.append(0.);for(int i=0;i<28;++i)c.knots.append(1.);s.curves.append(c);}

void scenario(const char *name,bool nurbs,double gap){

 const auto segments=forgeSketchSegments(sketch);const auto profile=buildProfile(segments,sketchConnectionTolerance());const auto axes=sketchAxes(sketch);const auto direction=forgeProjectionDirection(*target,sketch,false);

 std::vector<const ProfileLoop*> loops;for(const auto&r:profile.regions){loops.push_back(&r.outer);for(const auto &h:r.holes)loops.push_back(&h);}for(const auto &c:profile.chains)loops.push_back(&c);
 std::vector<CurveSpan> spans;std::vector<std::size_t> ends;for(const auto *loop:loops){for(const auto &seg:loop->segments)spans.push_back({embedCurve(seg.curve,axes),seg.range});ends.push_back(spans.size());}
 const auto projected=projectCurves(*target,spans,direction);double pieceMax=0.,chainMax=0.;std::size_t next=0;std::vector<ProjectedChain> chains;

 require(chains.size()==1&&app.size()==1,"fixture must join to one app and pipeline curve");require(std::isfinite(chainMax)&&chainMax<=ProjectionOptions().tolerance,"final chain exceeds default projection tolerance");
 double geometryError=0.;for(std::size_t j=0;j<chains.size();++j){auto a=std::dynamic_pointer_cast<const BSplineCurve<3>>(app[int(j)]);auto b=chains[j].curve;require(bool(a)&&bool(b),"expected spline output");require(a->degree()==b->degree()&&a->knots()==b->knots()&&a->weights()==b->weights()&&a->poles().size()==b->poles().size(),"app and pipeline spline structures differ");for(std::size_t k=0;k<a->poles().size();++k)require(distance(a->poles()[k],b->poles()[k])==0.,"app and pipeline poles differ");for(int k=0;k<=1024;++k){const auto d=b->domain();double t=d.lo+d.length()*k/1024.;geometryError=std::max(geometryError,distance(a->point(t),b->point(t)));}}

}
