    decl=[r for r in refs if r['file']==header and not r['text'].startswith('//') and re.search(r'\b'+short+r'\s*\(',r['text'])]

    items.append(dict(symbol=symbol,declaration=decl[0],definition=defs[0],recommendation=decision,rationale=reason,related=related,graph_incoming=incoming,graph_outgoing=outgoing,lexical_references=refs,definition_graph_description=nodes[definition]['description']))
